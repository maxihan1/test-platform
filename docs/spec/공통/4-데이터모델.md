# 데이터 모델

> **이 장이 담는 절** — §6 전체 스키마
> 표는 도메인마다 주인이 있지만 **스키마는 한 곳에서 본다**. 박제 원칙과 계약 변경 보고문이 표를 가로질러 설명하기 때문이다. 쪼개면 관계가 안 보인다.
> 전체 목차와 작업별 읽기 동선은 [SPEC.md](../../SPEC.md).

---

## 6. 데이터 모델

```sql
-- 서비스 (2026-09-17 신설). 화면 이름 「애플리케이션」 — 사람이 주소로 들어가는 사이트 · 앱 하나. 접두사가 이 서비스의 케이스를 가른다 (§1 · §2)
-- 화면 없는 API 를 직접 시험하는 API 서버도 한 행이다 — kind 'API' (2026-10-20 · 아래 「API 테스트 표」)
CREATE TABLE service (
  id         BIGSERIAL PRIMARY KEY,
  project_id BIGINT      NOT NULL REFERENCES project(id),  -- 속한 프로젝트 (2026-10-19 · 아래 「프로젝트 표」). 안 주면 트리거가 같은 이름으로 만들어 붙인다
  prefix     TEXT        UNIQUE NOT NULL,  -- tcId 앞부분. ^[A-Z][A-Z0-9]{0,11}$ (§2). 배정한 뒤에는 바꾸지 않는다
  name       TEXT        NOT NULL,
  color      TEXT        NOT NULL,         -- 화면이 안 읽는다(2026-09-22). 규칙의 정본은 §8 이다
                                            -- 2026-09-21 에 띠에서 자리 넷 줄로 옮겼을 때 이 주석만 안 따라와 「맨 위 띠」로 남아 있었다. 사이드바로 옮기며 같이 고쳤다
  tests_repo TEXT        NOT NULL,         -- 이 서비스의 테스트 코드 저장소 주소. 기재만 하고 받아오지는 않는다
  tests_dir  TEXT        NOT NULL,         -- 플랫폼이 실제로 읽을 폴더. PLATFORM_TESTS_DIR 아래 상대경로 (§9.2)
  slack_webhook TEXT,                     -- 실행 완료 알림을 보낼 곳 (§8.9). 비면 이 서비스는 알림을 안 보낸다. 비밀값이라 화면에 되돌려 보여주지 않는다
  figma_token TEXT,                       -- 작성 에이전트가 피그마 자료를 읽을 열쇠 (2026-09-23). 비면 이 서비스의 피그마 자료를 못 읽는다. slack_webhook 과 같은 비밀값 규칙이다 — 칸의 규칙은 도메인/인증 §8.8
  crawl_exclude TEXT[] NOT NULL DEFAULT '{}',  -- 역방향 작성(화면만 · 대조)이 훑지 않을 경로 목록 (2026-10-04 · 마이그레이션 20261004000001). 값 규칙은 도메인/인증 §7 계약 변경 블록
  -- 정기 실행 (2026-10-12 · 마이그레이션 20261012000001). 아래 처음 셋(schedule_at · schedule_tz · schedule_env)이 비면 꺼짐이다. 돌 조건은 공통/6-인프라 §9.2, 값 규칙은 도메인/인증 §7 계약 변경 블록
  schedule_at   TIME,                     -- 매일 돌릴 시각. 아래 시간대 기준
  schedule_tz   TEXT,                     -- 켠 사람 브라우저의 시간대 이름 (pg_timezone_names). 다시 저장해도 안 바뀐다
  schedule_env  TEXT,                     -- 대상 서버 키 (service_env.env). 외래키는 안 건다 — 대상 서버 줄은 저장마다 지우고 다시 넣는다
  schedule_last TIMESTAMPTZ,              -- 마지막으로 돈 때. 「오늘 돌았나」와 「마지막 정기 실행」 링크가 이것을 본다. 꺼도 남는다
  CHECK ((schedule_at IS NULL) = (schedule_tz IS NULL) AND (schedule_at IS NULL) = (schedule_env IS NULL)),
  is_active  BOOLEAN     NOT NULL DEFAULT true,   -- 지우지 않는다. 지우면 그 서비스로 돌린 과거 증적이 흔들린다
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 서비스마다 다른 대상 서버 주소 (§9). 환경변수 한 줄이던 것을 표로 올렸다 —
-- 서비스가 셋이면 주소가 아홉 개라 한 줄에 안 들어간다
CREATE TABLE service_env (
  service_id BIGINT NOT NULL REFERENCES service(id),
  env        TEXT   NOT NULL,   -- dev | qa | staging … 값은 조직이 정한다
  base_url   TEXT   NOT NULL,
  PRIMARY KEY (service_id, env)
);

-- 카탈로그 (스캔 결과 캐시. 진실의 원천은 코드)
CREATE TABLE test_case (
  tc_id           TEXT PRIMARY KEY,
  name            TEXT        NOT NULL,
  platforms       JSONB       NOT NULL DEFAULT '["desktop"]',   -- desktop | mobile | android. android 는 다른 값과 섞지 않는다 (2026-10-07)
  precondition    JSONB       NOT NULL DEFAULT '[]',
  file_path       TEXT        NOT NULL,
  param_schema    JSONB       NOT NULL,
  expected_schema JSONB       NOT NULL,
  is_active       BOOLEAN     NOT NULL DEFAULT true,
  scanned_at      TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 저장된 입력값 묶음 (파라미터별 행 분리 금지. JSONB 통째로)
CREATE TABLE param_set (
  id          BIGSERIAL PRIMARY KEY,
  tc_id       TEXT        NOT NULL REFERENCES test_case(tc_id),
  name        TEXT        NOT NULL,
  params      JSONB       NOT NULL,
  expected    JSONB       NOT NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  UNIQUE (tc_id, name)
);

-- 실행 묶음 (한 번의 실행 전체에 걸린 사실은 전부 여기에 박제한다)
CREATE TABLE test_run (
  run_id            BIGSERIAL PRIMARY KEY,
  title             TEXT        NOT NULL,
  service_id        BIGINT      REFERENCES service(id),   -- 어느 서비스의 실행인가. NULL은 서비스 통합 이전 행
  service_name      TEXT        NOT NULL,   -- 스냅샷. 증적 머리말에 찍힌다. 서비스 이름이 바뀌어도 과거 문서는 그대로
  tests_repo        TEXT        NOT NULL,   -- 스냅샷. 무엇을 시험했는지. 서비스 설정에서 받아 기재만 한다 (§8.8)
  triggered_by      TEXT        NOT NULL,   -- 로그인한 사람의 아이디 (§7 Auth). app_user를 가리키지만 FK는 걸지 않는다
  triggered_by_name TEXT,                   -- 스냅샷. 실행 당시의 사람 이름. NULL이면 인증 도입 이전 (§7.1)
  env               TEXT        NOT NULL,   -- 대상 서버 키. 기본값을 두지 않는다 (2026-09-16 결정)
  base_url          TEXT        NOT NULL,   -- 스냅샷. 그날 실제로 친 주소. env→주소 매핑이 바뀌어도 남는다
  status            TEXT        NOT NULL,   -- RUNNING | FINISHED | ABORTED
  notify_slack      BOOLEAN     NOT NULL DEFAULT false,  -- 끝나면 Slack 으로 알리기로 했나 (§8.9). 사람은 고르고 스케줄러는 늘 true
  notified_at       TIMESTAMPTZ,            -- 실제로 보낸 시각. notify_slack 이 true 인데 여기가 비었으면 보내려다 실패한 것이다
  started_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  finished_at       TIMESTAMPTZ,
  origin            TEXT        NOT NULL DEFAULT 'MANUAL',  -- 시작 방식 MANUAL | SCHEDULED | RERUN | BUILD | SINGLE (2026-10-19 · BUILD 는 2026-10-24 · SINGLE 은 2026-10-30 · 아래 「실행 시작 방식」)
  rerun_of          BIGINT      REFERENCES test_run(run_id) ON DELETE SET NULL, -- RERUN 이면 실패를 다시 돌린 원래 실행, SINGLE 이면 붙은 실행(없으면 NULL). 그 밖은 NULL
  -- 서비스 칸을 반만 채우고 INSERT 하는 것을 막는다. 둘 다 비었으면 통합 이전 행, 둘 다 찼으면 정상
  CHECK ((service_id IS NULL) = (service_name = ''))
);

-- 로그인 계정 (§3.5). 2026-09-17 에 등급 칸이, 2026-09-28 에 대시보드 권한 칸이 생겼다 — 아래 「권한 칸」 문단 참조
-- 프로젝트마다의 권한(케이스·실행·작성)은 아래 user_project 에 있다 (2026-10-19 PRJ-F1-02b — 앞은 서비스마다의 user_service 였다)
CREATE TABLE app_user (
  username      TEXT PRIMARY KEY,
  display_name  TEXT        NOT NULL,
  password_hash TEXT,                                    -- maxihan 계정(SSO)으로 만들어진 계정은 NULL 이다 (2026-10-20, 도메인/인증 §3.5 「maxihan 계정」)
  role          TEXT        NOT NULL DEFAULT 'member'  CHECK (role IN ('member','admin')),                 -- (§3.5)
  perm_dashboard TEXT       NOT NULL DEFAULT 'none'    CHECK (perm_dashboard IN ('none','read')),     -- 사람마다 하나. read 까지만
  is_approved   BOOLEAN     NOT NULL DEFAULT false,      -- 회원가입은 false 로 들어온다. admin 이 수락하면 true (§3.5)
  must_change_password BOOLEAN NOT NULL DEFAULT true,    -- 남이 정한 비밀번호면 true. 본인이 바꾸면 false (§3.5)
  is_active     BOOLEAN     NOT NULL DEFAULT true,       -- 지우지 않는다. 지우면 과거 실행의 실행자가 흔들린다
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  agent_token_hash TEXT,                                 -- 에이전트 토큰의 SHA-256 (2026-09-23, 도메인/인증 §7). 작성 에이전트 계정만 찬다. 원문은 안 둔다
  auth_sub      TEXT UNIQUE,                             -- Auth_platform 사용자 id(sub, UUID). maxihan 계정으로 묶이면 찬다 (2026-10-20, 도메인/인증 §3.5 「maxihan 계정」)
  farm_token    TEXT                                     -- 통합계정 토큰 원문(Auth_platform pat_… — 화면 이름 「통합계정 토큰」, 2026-10-31 앞 이름 「디바이스 팜 토큰」). 본인이 넣는다. 이 계정이 누른 실행이 폰을 빌릴 때와 이 계정이 누른 BTS 자동 설정에 쓴다 (2026-10-29, 도메인/러너 §5.2 「팜 토큰은 누른 사람 것이 먼저다」 · 도메인/BTS 「열쇠 표」)
);
CREATE UNIQUE INDEX app_user_agent_token_hash ON app_user (agent_token_hash) WHERE agent_token_hash IS NOT NULL;
REVOKE ALL ON app_user FROM grafana_ro;   -- 비밀번호 해시가 대시보드로 새면 안 된다

-- 계정이 어느 프로젝트에서 무엇을 하는가 (§3.5). 2026-09-28 에 권한 칸 셋이 생겼다 — 행이 있으면 배정이다.
-- 2026-10-19 PRJ-F1-02b 에서 주는 단위가 서비스(user_service)에서 프로젝트가 됐다.
-- 계정은 지우지 않고 비활성으로만 두어 계정 쪽 삭제는 막힐 일이 없다. 프로젝트는 애플리케이션을 옮겨 비면 설정 API 가 지우므로
-- 그 프로젝트의 권한 줄이 같이 가게 ON DELETE CASCADE 를 건다 (아래 「프로젝트 표」)
CREATE TABLE user_project (
  username   TEXT   NOT NULL REFERENCES app_user(username),
  project_id BIGINT NOT NULL REFERENCES project(id) ON DELETE CASCADE,
  perm_cases     TEXT NOT NULL DEFAULT 'none' CHECK (perm_cases IN ('none','read','write')),
  perm_runs      TEXT NOT NULL DEFAULT 'none' CHECK (perm_runs IN ('none','read','write')),
  perm_authoring TEXT NOT NULL DEFAULT 'none' CHECK (perm_authoring IN ('none','read','write')),
  -- 셋 다 none 인 행은 배정하지 않은 것과 같다. 두면 애플리케이션 고르개에 떠도 아무것도 못 한다 (§3.5)
  CONSTRAINT user_project_perm_check CHECK (perm_cases <> 'none' OR perm_runs <> 'none' OR perm_authoring <> 'none'),
  PRIMARY KEY (username, project_id)
);

-- 실행 항목 (항목마다 달라지는 사실만. 증적 한 줄을 만드는 데 필요한 것은 전부 실행 시점 스냅샷이다)
CREATE TABLE run_item (
  history_id      BIGSERIAL PRIMARY KEY,
  run_id          BIGINT      NOT NULL REFERENCES test_run(run_id),
  tc_id           TEXT        NOT NULL,
  platform        TEXT        NOT NULL DEFAULT 'desktop',  -- desktop | mobile | android (2026-10-07 — 값 제한이 없어 칸은 안 바뀐다)
  attempt         INTEGER     NOT NULL DEFAULT 1,   -- 회차. 같은 케이스×디바이스를 반복 실행한 순번 (1부터)
  tc_name         TEXT        NOT NULL,   -- 스냅샷. 이름이 바뀌어도 증적의 검증 항목 칸은 그대로
  file_path       TEXT        NOT NULL,   -- 스냅샷. 케이스가 다른 파일로 옮겨가도 무엇을 돌렸는지 남는다
  precondition    JSONB       NOT NULL DEFAULT '[]',   -- 스냅샷. 증적의 사전조건 칸
  params          JSONB       NOT NULL,   -- 스냅샷. 증적의 입력 칸 '값'. 비어 박제된 칸은 param_schema 의 default 로 보인다 (리포팅 §3.3)
  expected        JSONB       NOT NULL,   -- 스냅샷. 증적의 기대결과 칸 '값'. 빈 칸은 위와 같다
  param_schema    JSONB       NOT NULL,   -- 스냅샷. 증적의 입력 칸 '라벨'. 카탈로그는 캐시라 못 믿는다. 기본값 없음
  expected_schema JSONB       NOT NULL,   -- 스냅샷. 증적의 기대결과 칸 '라벨'. 기본값 없음
  param_set_id    BIGINT,                 -- 어느 입력값 묶음에서 왔는지. FK를 걸지 않는다 (묶음 삭제가 막히면 안 된다)
  timeout_ms      INTEGER,                -- 스냅샷. 이 항목에 몇 초를 줬는지. NULL은 마이그레이션 이전 행뿐이다 (아래 CHECK가 강제한다)
  status          TEXT        NOT NULL,   -- PASS | FAIL | NA. 생성 시 NA, finished_at이 NULL이면 아직 실행 전
  duration_ms     INTEGER,
  error           JSONB,
  started_at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  finished_at     TIMESTAMPTZ
);
CREATE INDEX ON run_item (run_id);
CREATE INDEX ON run_item (tc_id, platform, started_at DESC);  -- 케이스×디바이스 이력 추적
-- 같은 run 안에서 같은 케이스가 디바이스별·회차별로 각각 1행씩 쌓인다
CREATE UNIQUE INDEX ON run_item (run_id, tc_id, platform, attempt);
-- 마이그레이션 이전 행(file_path = '')과 INSERT 누락을 가른다. 둘 다 NULL이면 버그와 역사가 안 구분된다
ALTER TABLE run_item ADD CONSTRAINT run_item_timeout_recorded
  CHECK (file_path = '' OR timeout_ms IS NOT NULL);

-- 절차 + 검증 문장
CREATE TABLE run_item_step (
  id              BIGSERIAL PRIMARY KEY,
  history_id      BIGINT  NOT NULL REFERENCES run_item(history_id) ON DELETE CASCADE,
  seq             INTEGER NOT NULL,
  title           TEXT    NOT NULL,
  status          TEXT    NOT NULL,
  duration_ms     INTEGER,
  assertions      JSONB   NOT NULL DEFAULT '[]',  -- AssertionResult[]
  line            INTEGER,                        -- 실패한 소스 줄 번호
  screenshot_path TEXT,
  http_trace      JSONB,                          -- API 테스트의 요청·응답 원문
  error           JSONB,
  UNIQUE (history_id, seq)
);

-- 생성된 증적 문서. 「만드는 중」도 행으로 남긴다 — 그래야 새로고침해도 두 번 만들어지지 않는다
CREATE TABLE evidence_document (
  id           BIGSERIAL PRIMARY KEY,
  run_id       BIGINT      NOT NULL REFERENCES test_run(run_id),
  format       TEXT        NOT NULL,   -- PDF | XLSX | HTML (§8.4)
  status       TEXT        NOT NULL,   -- PENDING | READY | FAILED (§8.4)
  file_path    TEXT,                   -- READY일 때만 찬다
  error        TEXT,                   -- FAILED일 때 사람이 읽을 한 문장. 원문 오류를 그대로 쓰지 않는다
  generated_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  finished_at  TIMESTAMPTZ
);
-- 같은 실행·같은 형식으로 동시에 만드는 중인 것은 하나뿐이다. 버튼 연타와 새로고침을 DB가 막는다
CREATE UNIQUE INDEX ON evidence_document (run_id, format) WHERE status = 'PENDING';

-- 작성 대기줄 (2026-09-22 신설). 화면이 한 행을 넣고, 작성 에이전트(서버의 author 컨테이너 · 맥은 대체)가 집어 간다.
-- 에이전트는 러너처럼 같은 볼륨을 보지 않는다 — 오가는 것이 전부 이 서버(HTTP)를 지난다.
-- 글은 이 표로, 기획서 파일은 authoring_asset 과 자료 폴더로 (2026-09-23. 앞 판은 「전부 이 표를 지난다」였다)
CREATE TABLE authoring_request (
  id                BIGSERIAL PRIMARY KEY,
  service_id        BIGINT      NOT NULL REFERENCES service(id),   -- 한 요청은 한 서비스다. 접두사도 테스트 저장소도 여기서 나온다
  kind              TEXT        NOT NULL,   -- AUTHOR | RERUN | MERGE | EDIT(케이스 고치기, 2026-10-01 — 도메인/작성 §3.6 「★ 케이스 고치기」) (+ PRD_DRAFT · NOTE · BUGFIX 대기 — 아래 「BTS 연동 표」). 모두 한 줄에 선다 — 집어 가는 쪽이 작성 에이전트 하나라 줄도 하나다(동시 상한 안에서 같은 서비스도 나란히 · 반영은 서비스마다 차례로 · 도메인/작성 §3.6)
  source_id         BIGINT      REFERENCES authoring_request(id),  -- RERUN·MERGE 가 무엇에 대한 것인가. AUTHOR · EDIT(뿌리)는 비운다
  spec_text         TEXT,                   -- 옛 AUTHOR 행과 MERGE 행(「머지 요청 — 원본 #N」 자리표시)만 찬다. 2026-09-23 부터 기획서는 자료(authoring_asset)로 온다. 아래 「작성 자료」
  params            JSONB       NOT NULL DEFAULT '{}',   -- 케이스 고치기(EDIT)와 그 다시 적용(RERUN)은 { edits: [...] } (2026-10-01). 반영 요청(AUTHOR)과 그 재실행은 { prdApply: true } — 도메인/작성 §7 「표준 기획서 통로」 apply. 그 밖은 아직 안 쓴다
  requested_by      TEXT        NOT NULL,   -- 누가 눌렀나 (§7 Auth). app_user를 가리키지만 FK는 걸지 않는다
  requested_by_name TEXT        NOT NULL,   -- 스냅샷. 누를 당시의 사람 이름. test_run.triggered_by_name 과 같은 이유다
  claimed_by        TEXT,                   -- 집어 간 계정. app_user를 가리키지만 FK는 걸지 않는다 (requested_by와 같은 이유)
  status            TEXT        NOT NULL,   -- DRAFT | PENDING | RUNNING | DONE | FAILED (+ STOPPED, 2026-09-27 — 아래 「작성 중단 · 폐기 · 진척 칸」). DRAFT 는 자료를 올리는 중이라 아직 줄에 안 섰다
  stage             TEXT,                   -- 위층 진행 한 줄. 「케이스 2건째」·「관문 3 도는 중」. 아래층은 여기 안 들어온다
  stage_at          TIMESTAMPTZ,            -- 에이전트의 마지막 신호 시각(2026-09-27 부터 30초마다 — 처음엔 한 줄이 바뀐 시각). 「도는 중」과 「거기서 맥이 죽었다」를 가른다
  result            JSONB,                  -- 맥이 올린 판정. 러너와 같은 리포터를 타므로 §5.1 의 모양 그대로다
  test_source       JSONB,                  -- 스냅샷. 파일 경로 → 소스 본문. 머지 전 파일은 admin의 /tests 에 없다
  screenshot_dir    TEXT,                   -- 이 요청의 사진이 들어간 폴더. 대기줄 행 번호로 가른다 (2026-09-22 결정)
  pr_url            TEXT,                   -- 에이전트 스크립트가 연 초안 PR(자식 Claude 가 아니다, 작성.md §3.6). MERGE 요청은 이것을 보고 친다
  error             TEXT,                   -- FAILED일 때 사람이 읽을 한 문장. 원문 오류를 그대로 쓰지 않는다
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  started_at        TIMESTAMPTZ,            -- 맥이 집어 간 시각
  finished_at       TIMESTAMPTZ,
  -- 가리키는 것 없이 RERUN·MERGE 가 들어오거나, AUTHOR가 남의 행을 가리키는 것을 막는다
  -- 2026-10-01 이름을 붙여 다시 걸었다(authoring_request_source_pair_check · authoring_request_kind_check) — EDIT 도 뿌리다
  CHECK ((kind IN ('AUTHOR', 'EDIT')) = (source_id IS NULL)),
  -- 값 자체를 가둔다. 위 CHECK 는 짝만 보므로 {kind:'MERGE', source_id:42} 를 그냥 통과시킨다 —
  -- 오타난 kind 가 대기줄에 서면 맥이 「모르는 일」을 집는다 (2026-09-22 검토가 잡았다)
  CHECK (kind IN ('AUTHOR', 'RERUN', 'MERGE', 'EDIT')),
  -- 상태도 값 자체를 가둔다. 모르는 상태가 들어오면 집기도 화면 거르기도 그 행을 조용히 빠뜨린다
  CHECK (status IN ('DRAFT', 'PENDING', 'RUNNING', 'DONE', 'FAILED'))
);
-- 집기가 「그 서비스의 가장 오래된 대기 중」 한 건을 찾는 길. 줄이 길어져도 훑지 않는다
CREATE INDEX authoring_request_queue_idx ON authoring_request (service_id, status, id);

-- 작성 자료 (2026-09-23 신설). 작성 요청 하나에 딸린 기획서 파일과 피그마 주소. 여럿이 한 세트다
CREATE TABLE authoring_asset (
  id         BIGSERIAL PRIMARY KEY,
  request_id BIGINT      NOT NULL REFERENCES authoring_request(id) ON DELETE CASCADE,
  position   INTEGER     NOT NULL,   -- 넣은 순서. 맥이 이 순서대로 읽는다
  kind       TEXT        NOT NULL,   -- FILE | FIGMA (2026-10-20 API_DOC 이 는다 — 아래 「API 테스트 표」)
  name       TEXT        NOT NULL,   -- 화면에 보일 이름. FILE 은 사람이 준 파일 이름, FIGMA 는 figma_url 과 같다. 디스크 이름으로 쓰지 않는다
  figma_url  TEXT,                   -- FIGMA 만 찬다. 사람이 준 주소를 정규화해 다시 조립한 것 (도메인/작성 §7)
  size       BIGINT,                 -- FILE 만 찬다. 바이트
  created_at TIMESTAMPTZ NOT NULL DEFAULT now(),
  CHECK (kind IN ('FILE', 'FIGMA')),
  -- 파일인데 주소가 있거나 피그마인데 주소가 없으면 맥이 무엇을 읽을지 모른다
  CHECK ((kind = 'FILE') = (figma_url IS NULL)),
  -- 동시에 두 번 올려도 순서 번호가 겹치지 않게 DB 가 막는다
  UNIQUE (request_id, position)
);
```

**왜 이렇게까지 박제하는가** (2026-09-16 결정).
원칙은 하나다. **같은 실행의 증적은 언제 뽑아도 같은 문서가 나온다.** (예외 — E2E 요약 절, 정본은 도메인/리포팅 §3.3 「증적 재현」)
증적을 실행 직후에 뽑지 않으면 그 사이의 코드 변경이 문서에 섞여 들어간다.

**한 번의 실행 전체에 걸린 사실은 `test_run`에, 항목마다 다른 사실은 `run_item`에 둔다.**
서비스 이름·테스트 저장소·대상 서버·대상 주소·실행자는 실행 하나에 하나뿐이라
항목마다 반복될 값이 아니다.

| 칸 | 어디에 | 여기 없으면 |
|----|-------|-----------|
| `service_name` | `test_run` | 서비스가 여럿인데 증적 머리말이 어느 서비스의 문서인지 말하지 못한다. 이름을 나중에 고쳐도 과거 문서는 그대로여야 한다 (§8.8) |
| `service_id` | `test_run` | 실행 기록 목록을 서비스로 거르려면 `run_item`의 `tc_id` 접두사까지 따라가야 한다. 이 한 칸이 그 조인을 없앤다 |
| `tests_repo` | `test_run` | "무엇을 시험한 것인가"가 문서에 남지 않는다. 서비스 설정에서 받아 기재만 한다. 플랫폼이 저장소에서 직접 받아오지는 않는다 |
| `env` · `base_url` | `test_run` | `env`→주소 대응표는 `service_env`가 갖고 있고 설정은 바뀐다. 주소까지 박아야 반년 뒤에 "그때 QA가 어디였는지"를 되짚을 수 있다 |
| `triggered_by_name` | `test_run` | 계정 이름을 바꾸거나 계정을 지우면 과거 증적의 실행자 칸이 흔들린다 |
| `param_schema`·`expected_schema` | `run_item` | 입력·기대결과 칸의 **라벨**을 카탈로그에서 읽게 된다. 케이스 코드의 설명 문구(`.describe()`)를 고친 날, 반년 전 실행의 증적 라벨까지 같이 바뀐다 |
| `timeout_ms` | `run_item` | 타임아웃으로 `NA`가 난 항목을 나중에 볼 때 몇 초를 줬는지 알 수 없다. 5초를 준 것과 5분을 준 것은 전혀 다른 이야기다 |
| `param_set_id` | `run_item` | "어느 입력값 묶음으로 돌렸는지"가 사라진다. **외래키를 걸지 않는다** — 묶음을 지웠다고 과거 실행이 지워지거나 삭제가 막히면 안 된다 |
| `file_path` | `run_item` | 케이스 파일이 옮겨가거나 지워지면 그날 무슨 파일을 돌렸는지 되짚을 수 없다 |
| `attempt` | `run_item` | 반복 실행한 회차가 서로 덮어써진다 |
| `status`·`error` | `evidence_document` | 「만드는 중」을 둘 자리가 없어 새로고침할 때마다 또 눌리고 문서 행이 쌓인다. §3.3이 리포팅의 쓰기를 이 표로만 막아 둬서 다른 데 둘 수도 없다(2026-10-11 「버그」 판정 `run_case_bug` 가 둘째 표로 들었다) |
| `notify_slack`·`notified_at` | `test_run` | 한 칸만 두면 **보낼 생각이 없었던 것**과 **보내려다 실패한 것**이 구분되지 않는다. 「안 왔는데?」를 되짚을 수 없다 (§8.9) |
| `spec_text` | `authoring_request` | 기획서 경로만 남는다. **맥은 다른 기계라 그 경로를 못 읽고**, 그 파일이 고쳐지면 무엇을 시킨 요청이었는지도 같이 바뀐다. **2026-09-23 — 담는 자리가 이 칸에서 `authoring_asset` 과 자료 폴더로 옮겼다. 기준은 그대로다** (경로가 아니라 내용을 서버에 둔다). 왜 옮겼는지는 도메인/작성 §7 「자료」 |
| `requested_by_name` | `authoring_request` | 계정 이름을 바꾸거나 계정을 비활성으로 두면 지난 요청의 「누가 눌렀나」가 흔들린다. `test_run`과 같은 이유다 |
| `test_source` | `authoring_request` | 화면이 실패한 줄을 발췌하려면 소스 본문이 있어야 하는데 **머지 전 파일은 admin의 `/tests` 에 없다**. 받아 두지 않으면 머지된 뒤에야 발췌가 된다 |
| `stage_at` | `authoring_request` | 한 칸만 두면 **아직 그 단계를 도는 중**인 것과 **거기서 맥이 죽은 것**이 구분되지 않는다. 2026-09-27 부터 에이전트가 한 건 내내 30초마다 신호를 보내 「마지막 신호」가 됐다 — 3분 넘게 묵으면 죽은 것이다(도메인/작성 §7 「중단 · 폐기 · 진척」) |

**`test_run`에 서비스 칸을 두는 것은 좁은 예외다.** 서비스 통합을 정할 때 「모든 표에 어느 서비스 칸을
더할 필요가 없다 — 번호가 이미 서비스를 말한다」고 했다(§1). 그 말은 그대로다.
`test_case`·`run_item`·`param_set`에는 서비스 칸이 없고 앞으로도 없다.
`test_run`만 예외인 이유는 둘이다 — **한 실행은 한 서비스**라는 불변식을 걸 자리가 필요하고(§8.2),
증적 머리말이 **그때의 서비스 이름**을 박제해야 하기 때문이다.

`env`에 기본값을 두지 않는 이유. 안 채우면 빈 칸이 아니라 **틀린 값**이 남는다.
`DEFAULT 'demo'`가 있으면 QA 서버에 쏜 실행도 증적에 `demo`로 찍혀 문서가 거짓말을 한다.
지금 실행 생성 코드가 실제로 `env`를 넣지 않고 INSERT 하고 있어 이미 그 상태다.
`timeout_ms`에 기본값을 두지 않고 NULL을 허용하는 것도 같은 이유다.
`300000`을 채워 두면 빈 칸이 아니라 틀린 숫자가 남는다.

**마이그레이션 이전에 쌓인 행에는 이 원칙이 소급되지 않는다.**
새로 생긴 칸은 옛 행에서 `''`·`'{}'`·`NULL`로 남는다.
화면과 증적 문서는 이것을 **`기록 없음`** 으로 쓰고 **채워 넣지 않는다.**
모르는 것을 아는 척하면 그게 다음 거짓말이다.

**권한 칸(`role`)을 만들지 않는다** (2026-09-16 결정). 권한 구분·팀·승인 절차는 지금 하지 않는다.
쓰지도 않을 칸을 미리 만들면 전부 같은 값이 들어가고, 그 칸을 보는 코드가 생기면서
"구분이 있는 것처럼" 보이기 시작한다. 구분이 실제로 필요해지는 날 칸을 더한다.
같은 이유로 `is_active`도 만들지 않는다.

> **2026-09-17 — 위 결정을 뒤집었다.** `role`과 `is_active`를 둘 다 만든다.
> **문단을 지우지 않고 남기는 이유는 판단 기준이 그대로 살아 있기 때문이다.** 다음에 칸을 하나 더
> 늘리고 싶을 때 다시 이 문단을 읽고 같은 질문을 해야 한다 — "이 칸을 지금 쓰는 코드가 있는가."
>
> | 칸 | 무엇이 「필요해지는 날」을 데려왔나 |
> |----|--------------------------------|
> | `role` | 서비스와 계정을 **화면에서** 만들게 됐다(§8.8). 그 화면을 아무나 열면 안 된다. 권한의 정본은 §3.5 다 |
> | `is_active` | 서비스도 계정도 **지우지 않기로** 했다. 지우면 그것으로 돌린 과거 실행의 증적이 흔들린다 |
>
> **`role`에는 기본값을 둔다** (`'viewer'`). `env`에 기본값을 두지 않는 것과 반대로 보이지만
> 기준은 같다 — **빠뜨렸을 때 틀린 값이 남는가.** `env`의 `'demo'`는 QA에 쏜 실행을 `demo`라고
> 거짓말한다. `role`의 `'viewer'`는 **가장 좁은 등급**이라, 빠뜨리면 권한이 모자라 그 자리에서 드러난다.
> 조용히 틀리지 않는다.
>
> **전제 하나가 깔려 있다** — 「보기만」 등급인 사람이 실제로 존재해야 한다. 아니면 이 문단이
> 경고한 상황 그대로가 된다. 그 전제는 §3.5에 적어 뒀다.

> **2026-09-28 — 등급 셋을 기능별 권한으로 바꿨다** (§3.5). `role` 은 `member | admin` 둘로 줄고
> **서비스마다** 칸 셋(`user_service.perm_cases` · `perm_runs` · `perm_authoring`)과 **사람마다** 대시보드 칸 하나(`app_user.perm_dashboard`)가 생겼다.
> 같은 날 처음 판은 넷을 다 `app_user` 에 뒀다 — 같은 사람이 서비스마다 다른 일을 한다는 요청으로 코드가 서기 전에 옮겼다.
> 같은 날 `is_approved`(회원가입 승인 대기)와 `must_change_password`(비밀번호 변경 강제)도 생겼다.
> **위 질문 「이 칸을 지금 쓰는 코드가 있는가」에 칸마다 답이 있다** — 권한 칸은 문(`auth/gate.ts`)이,
> `is_approved` 는 로그인이, `must_change_password` 는 문이 읽는다.
>
> **기능마다 칸을 두고 `(username, service_id, feature, level)` 표를 따로 만들지 않았다.** 그런 표는 기능이 늘 때
> 마이그레이션 없이 늘릴 수 있지만, **기능이 느는 날은 문의 표와 화면도 같이 바뀌는 날**이라 마이그레이션 하나가
> 더 드는 것은 비용이 아니다. 대신 칸이면 **`CHECK` 가 값을 가둔다** — 대시보드에 `write` 가, 셋 다 `none` 인 배정이 들어갈 수 없다.
>
> **기본값은 전부 가장 좁은 쪽이다** — 권한 칸 `'none'` · `is_approved` `false` · `must_change_password` `true`.
> 위 `'viewer'` 와 같은 기준이다 — **빠뜨리면 모자라서 그 자리에서 드러난다.** `must_change_password` 를 빠뜨린 가입은
> 첫 로그인에서 한 번 더 바꾸라는 화면이 뜰 뿐이고, 거꾸로 `false` 가 기본이면 임시 비밀번호가 조용히 영구 비밀번호가 된다.
>
> **옮기는 마이그레이션**(`db/migrations/<날짜>_permissions.sql`)은 **기존 계정의 할 수 있는 일을 바꾸지 않는다** —
> 배정된 서비스 행마다 `viewer` → 셋 다 `read` · `operator` → 셋 다 `write`, 두 등급 다 대시보드 `read` ·
> `admin` → `admin`(서비스 행의 셋은 `write`, 대시보드 `read` 로 채운다 — admin 은 칸을 안 보지만, member 로 내렸을 때 할 일이 줄지 않게).
> `user_service` 에 칸을 더할 때 `DEFAULT 'read'` 로 채워 셋-다-none `CHECK` 를 지나게 한 뒤 옮기고 기본값을 `'none'` 으로 바꾼다. 기존 계정은 `is_approved = true` · `must_change_password = false` 로 채운다 —
> 이미 쓰던 사람을 가입 대기로 돌리거나 다시 바꾸게 하지 않는다. **값은 칸을 더할 때 `DEFAULT` 로 채우고, 그다음에 기본값을 좁은 쪽으로 바꾼다.**
> 대조의 정본은 도메인/인증 §7 「등급으로 갈리는 자리」 표 아래 「옛 등급에서 옮긴 값」이다.

### 작성 대기줄 (`authoring_request`, 2026-09-22 신설)

**작성 에이전트와 주고받는 유일한 자리다.** 화면이 행을 넣고 **작성 에이전트**(서버의 author 컨테이너 · 맥은 대체)가 집어 간다.
옆 컨테이너 러너는 어드민과 같은 볼륨을 봐서 경로만 주고받으면 되지만(볼륨을 같이 안 보는 러너는 admin 이 받아 온다 — 공통/3 §5.3) **맥은 다른 기계다**(§9.2의 정기 실행이
컨테이너 안 명령으로 풀린 것과 다른 점이다 — 2026-10-12 부터는 admin 이 안에서 만든다). 그래서 테스트 소스 본문은 이 표를, 기획서는
**자료 표(`authoring_asset`)와 자료 폴더**를 지난다 — 아래 「작성 자료」.
2026-09-23 전에는 「기획서 본문도 이 표를 지난다」였다. 붙여 넣던 본문이 올리는 파일로 바뀌었다.

**집어 가는 일은 한 줄 UPDATE 다** — `PENDING`을 `RUNNING`으로 바꾸는 것이 곧 잠금이라
**잠그려고** 누가 집었는지를 적을 필요는 없다.

> **2026-09-22 — 다른 이유로 그 칸(`claimed_by`)을 만들었다. 위 문장은 그대로 맞다.**
> 잠금에는 여전히 필요 없다. 필요해진 것은 **귀속**이다 — 집은 쪽을 안 적으면
> `operator` 누구나 `finish`에 임의의 `pr_url`을 실을 수 있고, 뒤에 `admin`이 머지를 누르면
> **그 사람이 고른 PR이 병합된다.** 사람 게이트는 「이 요청을 머지할까」만 묻고
> **「이 `pr_url`이 그 요청이 만든 것 맞나」는 아무도 안 본다** (2026-09-22 검토가 잡았다).
>
> **문단을 지우지 않는 이유는 판단 기준이 살아 있기 때문이다** — 다음에 칸을 하나 더 늘리고 싶을 때
> 「잠금 때문인가, 다른 이유인가」를 다시 물어야 한다. 잠금 때문이면 여전히 필요 없다.

**진행은 두 층이고 이 표는 위층만 든다.**

| 층 | 무엇 | 어디에 |
|----|-----|-------|
| 아래 — **테스트 절차** | 「할 일을 한 건 만든다」 2/2 | **이미 있다.** kit이 절차마다 한 줄을 흘리고 러너가 모아 §5.1의 모양으로 낸다 |
| 위 — **작업 단계** | 「요구사항 표 만드는 중」 → 「케이스 2건째」 → 「관문 3 도는 중」 | `stage` 한 줄 |

아래층을 여기에 또 담으면 같은 것이 두 곳에서 다른 말을 한다. 아래층은 `result`에 통째로 실려 온다.

**서비스 칸은 `service_id` 하나뿐이고 이름은 박제하지 않는다.** 한 요청은 한 서비스다(§1).
`test_run`이 `service_name`까지 박제한 이유는 **증적 문서 머리말**이 그때의 이름을 찍어야 하기
때문인데, 작성 요청은 증적을 만들지 않는다. 화면은 그릴 때 서비스 이름을 다시 읽으면 된다.
**박제는 문서가 굳는 자리에만 건다.** 대신 `service_id`는 `NOT NULL`이다 — 새 표라
「통합 이전 행」이 없어서 `test_run`처럼 비워 둘 이유가 없다.

**작성·재실행·머지 셋이 한 줄에 선다.** `kind`로 가르고, 재실행과 머지는 `source_id`로 **무엇에 대한
것인지**를 가리킨다. `CHECK`가 그 짝을 강제해서, 가리키는 것 없는 머지 요청은 INSERT가 그 자리에서 실패한다.
**행은 실행 하나다** (2026-09-29) — 사람에게 보이는 요청 번호는 `source_id` 를 AUTHOR 까지 거슬러 오른 **뿌리** 하나이고,
재실행 · 이어받기 · 머지 행은 그 번호의 실행 기록이다. 표는 그대로다(도메인/작성 §7 「실행 기록」).

**이 표는 2026-09-22에 실제로 섰다** (`db/migrations/20260922000001_authoring_request.sql`).
그 마이그레이션이 지킨 것 둘.

- **`DEFAULT`를 넣었다 빼는 춤이 필요 없다.** `test_run`·`run_item`이 `NOT NULL DEFAULT ''`로 넣고
  곧바로 `DROP DEFAULT`한 것은 **이미 쌓인 행**을 채우려던 것이다. 새 표에는 채울 옛 행이 없다

  새 표는 자동으로 붙지 않는다. 대시보드가 이 표를 읽을 일이 생기면 **그때** `GRANT` 한 줄을 더한다.

### 작성 자료 (`authoring_asset`, 2026-09-23 신설)

**기획서는 붙여 넣는 글이 아니라 파일로 온다.** PDF·워드 파일이 여럿이고 피그마 화면도 같이 온다.
그 한 세트가 작성 요청 하나에 딸린 **자료**다. 파일 한 개가 한 행, 피그마 주소 한 개가 한 행이다.
**받는 종류·상한·주소 모양은 도메인/작성 §7 「자료」가 정본이다.** 여기 옮겨 적지 않는다.

- **파일은 디스크에, 이름은 표에.** 저장 자리는 `authoring-assets/<요청 번호>/<자료 번호>.<확장자>` 이고
  뿌리는 사진과 같은 `PLATFORM_ARTIFACTS_DIR` 다 — 볼륨으로 묶인 유일한 자리라서다.
  **디스크 이름은 자료 행 번호로 짓는다.** 사람이 준 이름(`기획서.pdf`)은 `name` 칸에만 둔다 —
  사람이 준 이름으로 파일을 만들면 `../` 같은 글자로 폴더 밖에 쓸 수 있다
- **사진 폴더(`authoring/<번호>/`) 안에 두지 않는다.** 그 폴더는 사진 장수를 세서 상한을 건다.
  같이 두면 자료까지 사진으로 센다

### 실행 종류 (`test_run.kind`, 2026-10-02 · PR #131)

**실행 하나에는 한 종류만 담는다** — `UI`(UI 테스트) · `FN`(기능 테스트) · `SCENARIO`(E2E). 케이스 실행은 `UI` · `FN` 둘이다.
종류를 가르는 뜻은 [도메인/작성](../도메인/작성.md) §3.6 「★ 테스트 두 갈래」, 목록 · 오류 모양은 [도메인/실행](../도메인/실행.md) §7 이 정본이다.

- **종류는 케이스 번호로 정한다** — `-UI-` 면 `UI`, 그 밖(옛 꼴 · `-FN-`)은 `FN`(`catalog/rules.ts` `tcId종류`). 섞이면 실행을 만들지 않는다(400 `MIXED_KIND`)
- **기본값은 `FN`** — 옛 꼴 번호를 기능으로 보는 규칙과 같다. 서버는 실행을 만들 때 늘 종류를 적는다. 기본값은 배포 틈(옛 admin 의 INSERT)과 손으로 넣는 행을 위한 것이다 (2026-10-02 게이트 1)
- **옛 `CASE` 실행은 지운다** (2026-10-02 사용자 — DEMO · TODO 포함). 증적 문서 → 실행 항목(절차 행은 CASCADE) → 실행 순이다.
  **되돌려도 돌아오지 않는다.** 디스크의 스크린샷 · 증적 파일은 남는다 — 지워도 되는 자리는 docs/SETUP.md
- **「케이스 실행」 조건 글자는 한 자리**(`apps/admin/src/execution/runKind.ts`)에 둔다. 코드에 `'CASE'` 를 다시 쓰지 않는다
- **견주기의 직전 실행은 같은 종류에서만 찾는다** — UI 실행 앞에 기능 실행을 두면 모든 케이스가 「빠졌다」로 뜬다. 스모크 아닌 실행은 스모크(`build_stage = 'SMOKE'`)를 건너뛴다(2026-10-25 GAME-F2-14 · 정본은 도메인/리포팅 §7 insights previous)
- **정해진 시간 전체 실행은 종류별로 실행 둘을 만든다.** 빈 종류는 안 만든다. 한쪽이 실패해도 다른 쪽은 돈다

### 실행 시작 방식 (`test_run.origin` · `rerun_of`, 2026-10-19 · ⑫ 사용성 개편 WEB-F4-01)

**실행이 어떻게 시작됐는지와 무엇을 다시 돌린 것인지를 실행에 남긴다.** 실행 기록이 「사람이 직접 건 것 · 정해진 시간에 돈 것 · 실패를 다시 돌린 것」을 가르고,
다시 돌린 실행이 원래 실행으로 이어진다. 칸을 쓰는 규칙(자동 제목 · 거르기)은 [도메인/실행](../도메인/실행.md) §7 `POST /api/runs` · `GET /api/runs` 와 §8.7 이 정본이다.

- **값은 다섯이다** — `MANUAL`(사람이 실행 창에서 걸었다) · `SCHEDULED`(정해진 시간 실행 `execution/scheduled.ts`) · `RERUN`(실행 결과의 「실패 N건 다시 실행」 — 반복 1 실행에서 1건이면 `SINGLE`) · `BUILD`(CI 가 빌드를 알렸다 — 2026-10-24 GAME-F1-13 · [도메인/실행](../도메인/실행.md) §3.11) ·
  `SINGLE`(케이스 하나 · 반복 1 — 2026-10-30 WEB-F7-02 · [도메인/실행](../도메인/실행.md) §3.2 「★ 한 건 실행은 묶음 실행에 붙는다」).
  화면 글자는 「직접」 · 「정기」 · 「실패 다시」 · 「빌드」다. `BUILD` 칸은 아래 「게임 테스트 표」 smoke 블록이 더했다.
  **`SINGLE` 은 실행 기록에 줄로 서지 않아 화면 글자가 없다** — 케이스 상세 「실행 이력」에서만 「한 건 실행」 · 「다시 실행 · RUN N 에 붙음」으로 보인다
- **`rerun_of` 는 `RERUN` · `SINGLE` 일 때만 찬다** — 그 밖인데 찬 행은 DB 가 막는다. `SINGLE` 은 붙은 실행이고 붙일 곳이 없으면 비어 있는 것이 정상이다(2026-10-30). 원래 실행이 지워지면 `NULL` 로 돌아간다(`ON DELETE SET NULL`) — 제품은 실행을 지우지 않지만
  DB 테스트 · 옛 `CASE` 정리처럼 `test_run` 을 지우는 코드가 있어 삭제를 막으면 안 된다. 그래서 「`RERUN` 이면 반드시 찬다」는 DB 가 걸지 않고 서버가 만들 때 지킨다. 화면은 `RERUN` 인데 비었으면 「원래 실행 없음」으로 적는다
- **옛 행은 실행자로 채운다** — 정해진 시간 실행은 실행자를 `스케줄러` 로 박제해 왔다(`triggered_by`). 그 행만 `SCHEDULED`, 나머지는 `MANUAL` 이다.
  옛 「실패 다시」 실행은 가를 재료가 없어 `MANUAL` 로 남는다 — 지어내지 않는다
- **시나리오 실행(`kind = SCENARIO`)도 같은 칸을 쓴다** — 지금은 늘 `MANUAL` 이다. 시나리오에는 정해진 시간 실행 · 실패 다시가 없다
- **실행 제목 칸(`title`)은 그대로다** — 새 실행은 서버가 지은 제목을, 옛 실행은 그때 저장한 제목을 들고 있다. 제목을 다시 짓지 않는다

```sql
ALTER TABLE test_run
  ADD COLUMN origin   TEXT   NOT NULL DEFAULT 'MANUAL',
  ADD COLUMN rerun_of BIGINT REFERENCES test_run(run_id) ON DELETE SET NULL;
UPDATE test_run SET origin = 'SCHEDULED' WHERE triggered_by = '스케줄러';
ALTER TABLE test_run ADD CONSTRAINT test_run_origin_check CHECK (origin IN ('MANUAL','SCHEDULED','RERUN'));
ALTER TABLE test_run ADD CONSTRAINT test_run_rerun_of_check CHECK (rerun_of IS NULL OR origin = 'RERUN');
```

**`SINGLE` 을 더한다** (2026-10-30 WEB-F7-02 — 계약 블록은 [도메인/실행](../도메인/실행.md) §3.2, 반영 완료 2026-10-30 PR #529 · 마이그레이션 `20261030000001_run_single`). 새 칸은 없다 — 붙은 실행은 `rerun_of` 가, 덮었는지는 붙은 실행의 항목과 (케이스, 디바이스)를 맞대 서버가 센다.
덮은 것을 칸으로 박제하지 않는 까닭 — 덮는 규칙(판정이 난 것만 · 가장 늦게 끝난 것)이 행 둘 사이의 관계라 칸에 적으면 둘이 어긋날 수 있다.

```sql
ALTER TABLE test_run DROP CONSTRAINT test_run_origin_check;
ALTER TABLE test_run ADD CONSTRAINT test_run_origin_check CHECK (origin IN ('MANUAL','SCHEDULED','RERUN','BUILD','SINGLE'));
ALTER TABLE test_run DROP CONSTRAINT test_run_rerun_of_check;
ALTER TABLE test_run ADD CONSTRAINT test_run_rerun_of_check CHECK (rerun_of IS NULL OR origin IN ('RERUN','SINGLE'));
CREATE INDEX test_run_single_of ON test_run (rerun_of) WHERE origin = 'SINGLE';   -- 실행마다 붙은 한 건을 찾는다
```

### 저장값 표 (`case_input`, 2026-09-29 · 게이트 0·1)

**뜻과 규칙은 도메인/실행 §8.2 「저장값 — 케이스마다 한 벌」, 실행을 만들 때 채우는 규칙은 같은 장 §3.2 가 정본이다.** 여기는 표만 적는다.

```sql
-- 케이스마다 한 벌 저장하는 실행 입력값. 이름 없이 케이스당 한 행이다
CREATE TABLE case_input (
  tc_id     TEXT        PRIMARY KEY REFERENCES test_case(tc_id),
  params    JSONB       NOT NULL,   -- 코드 기본값과 다른 칸만. 비밀값은 평문 (공통/2-명세선언 §4.1)
  expected  JSONB       NOT NULL,   -- 같다
  saved_by  TEXT        NOT NULL,   -- 누가 저장했나. 세션의 사람
  saved_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);
```

- **주인은 실행이다.** 쓰는 것은 실행의 저장 통로(도메인/실행 §7 「저장값」)뿐이다.
  **예외 하나 — 지우기** (2026-10-01): 케이스 고치기의 반영이 병합으로 끝나면 작성 쪽 끝내기가 바꾼 기대값 칸을 지운다. 함수는 실행 쪽(`execution/savedInput.ts`)에 있고 작성이 부른다 (도메인/작성 §3.6 「★ 케이스 고치기」)
  **카탈로그는 SQL 로 읽기만 한다** — 케이스 응답의 `savedInput` 을 채우려고 `LEFT JOIN` 한다(도메인/카탈로그 §3.1 · §7)
- **`param_set` 과 따로 둔다.** 묶음은 이름을 붙여 여러 벌이고 사람이 골라야 쓰인다. 저장값은 한 벌이고 고르지 않아도 실행에 쓰인다
- **코드 기본값과 같은 칸은 저장하지 않는다.** 칸이 남지 않으면 행을 지운다 — 기본값의 원천은 여전히 코드다 (공통/6-인프라 §9.2)
- `run_item` 은 이 표를 가리키지 않는다. 채운 값을 `params`·`expected` 에 복사해 둔다 (박제 원칙)
- 대시보드 계정에 열지 않는다 — 비밀값이 평문으로 든다(위 「대시보드 칸 권한」)

### 표준 기획서 · 지도 · 화면 기록 표 (2026-10-10)

### trace 기록 표 (`run_trace` · `run_network` · `run_action` · `run_console`, 2026-10-16 · PRD-F10-03)

정식 실행 항목의 trace.zip 을 푼 것이다. **무엇을 왜 남기고 어떻게 가리는지는 [도메인/실행.md](../도메인/실행.md) §3.2 「trace」가 정본이다.** 여기는 표와 칸만 적는다.

```sql
-- 항목마다 한 줄. 줄이 없으면 아직 안 풀었거나 이 기능 전의 항목이다
CREATE TABLE run_trace (
  history_id BIGINT      PRIMARY KEY REFERENCES run_item(history_id) ON DELETE CASCADE,
  status     TEXT        NOT NULL CHECK (status IN ('DONE', 'NONE', 'FAILED')),
  reason     TEXT,                                -- NONE · FAILED 의 사유 한 줄
  loaded_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 값은 모두 가린 뒤 넣는다. 헤더는 안 남긴다. 본문은 실패 항목의 API 줄만 2000자까지
CREATE TABLE run_network (
  id            BIGSERIAL PRIMARY KEY,
  history_id    BIGINT  NOT NULL REFERENCES run_item(history_id) ON DELETE CASCADE,
  ord           INTEGER NOT NULL,                 -- 그 항목 안의 차례
  seq           INTEGER,                          -- 단계 번호. 준비 · 정리 중이면 NULL
  at_ms         INTEGER NOT NULL,                 -- 테스트 시작부터
  method        TEXT    NOT NULL,
  url           TEXT    NOT NULL,
  status        INTEGER,                          -- 응답이 없으면 NULL
  resource_type TEXT    NOT NULL,                 -- document · fetch · xhr · api(요청 도구) · 그 밖은 실패한 것만
  duration_ms   INTEGER,
  failure       TEXT,
  request_body  TEXT,
  response_body TEXT,
  UNIQUE (history_id, ord)
);

CREATE TABLE run_action (
  id          BIGSERIAL PRIMARY KEY,
  history_id  BIGINT  NOT NULL REFERENCES run_item(history_id) ON DELETE CASCADE,
  ord         INTEGER NOT NULL,
  seq         INTEGER,
  at_ms       INTEGER NOT NULL,
  title       TEXT    NOT NULL,                   -- Playwright 가 붙인 이름(「Click getByRole(…)」)
  target      TEXT,                               -- locator · 주소
  value       TEXT,                               -- 넣은 값. 비밀번호 칸이면 가린다
  duration_ms INTEGER,
  error       TEXT,
  UNIQUE (history_id, ord)
);

CREATE TABLE run_console (
  id         BIGSERIAL PRIMARY KEY,
  history_id BIGINT  NOT NULL REFERENCES run_item(history_id) ON DELETE CASCADE,
  ord        INTEGER NOT NULL,
  seq        INTEGER,
  at_ms      INTEGER NOT NULL,
  level      TEXT    NOT NULL,                    -- error · warning · pageerror
  text       TEXT    NOT NULL,
  url        TEXT,
  UNIQUE (history_id, ord)
);
```

- **세 표에는 가린 값만 들어간다** — `run_item.params` 는 다시 실행하려고 평문으로 두지만(§8.2), 이 표들은 화면에 그대로 보일 기록이라 넣을 때 가린다
- **항목을 지우면 같이 지워진다**(`ON DELETE CASCADE`). 지우는 주기는 L6 와 같이 정한다([1-제품과-구조.md](1-제품과-구조.md) 「지금 만들지 않는다」)
- `grafana_ro` 에는 안 준다 — 어느 패널도 안 읽는다(「대시보드 칸 권한」은 패널이 읽는 칸에만 준다)

계약 블록은 [도메인/실행.md](../도메인/실행.md) §3.2 「trace」에 있다(표 넷 · 항목 상세 응답 · 경로 규칙을 한 블록에 적었다).

### 서비스 지도 표 (`service_api` · `service_api_use`, 2026-10-16 · PRD-F10-05)

대상 서비스가 부르는 같은 사이트 API 와 그 API 를 거친 케이스 · 화면이다. **무엇을 넣고 언제 갈아 끼우는지는 [도메인/작성.md](../도메인/작성.md) §3.6 「서비스 지도」가 정본이다.** 여기는 표와 칸만 적는다.

```sql
-- API 하나. 주소 틀은 숫자 · 주문번호 꼴 · uuid · 긴 16진 칸을 자리표로 바꾸고 물음표 뒤를 뺀 경로다 (apps/admin/src/prd/apiMap.ts `API틀`).
-- 주문번호 꼴은 2026-10-26 에 더했고 주문마다 갈려 쌓인 줄은 20261026000001_api_map_merge.sql 이 한 번 합쳤다(WEB-F6-03).
-- 같은 마이그레이션이 계정 아이디로 가려진 별표 칸 줄(/api/********/banners)도 지웠다 — 다음 정식 실행이 다시 쌓는다
CREATE TABLE service_api (
  id         BIGSERIAL   PRIMARY KEY,
  service_id BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  method     TEXT        NOT NULL,
  host       TEXT        NOT NULL,                -- 포트까지(localhost:3001). 대상 서버마다 다르면 다른 줄이다
  path       TEXT        NOT NULL,
  first_seen TIMESTAMPTZ NOT NULL DEFAULT now(),
  last_seen  TIMESTAMPTZ NOT NULL DEFAULT now(),
  -- 정식 실행 trace 가 본 마지막 2xx 코드. NULL 이면 성공 응답을 못 봤다. E2E 미룬 삭제가 그 코드로 답한다 (2026-10-16 · 도메인/시나리오 §3.7 결정 12)
  ok_status  INTEGER     CHECK (ok_status IS NULL OR ok_status BETWEEN 200 AND 299),
  UNIQUE (service_id, method, host, path)
);

-- 그 API 를 누가 거쳤나. 케이스 줄(정식 실행 trace)이거나 화면 줄(크롤러) 둘 중 하나다
CREATE TABLE service_api_use (
  id         BIGSERIAL   PRIMARY KEY,
  api_id     BIGINT      NOT NULL REFERENCES service_api(id) ON DELETE CASCADE,
  service_id BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  tc_id      TEXT,                                -- 케이스 줄
  run_id     BIGINT,                              -- 그 케이스 줄을 채운 실행. 더 새 실행을 풀면 그 케이스 줄을 갈아 끼운다. 실행을 지워도 지도는 남는다
  steps      INTEGER[]   NOT NULL DEFAULT '{}',   -- 그 API 를 부른 단계 번호. 준비 · 정리 중 호출은 안 든다
  env        TEXT,                                -- 화면 줄 — 화면 기록 열쇠(대상 서버 · 상태 · 같은 틀)
  state      TEXT,
  screen_url TEXT,
  seen_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  CHECK (
    (tc_id IS NOT NULL AND run_id IS NOT NULL AND env IS NULL AND state IS NULL AND screen_url IS NULL)
    OR (tc_id IS NULL AND run_id IS NULL AND env IS NOT NULL AND state IS NOT NULL AND screen_url IS NOT NULL)
  ),
  -- 화면 기록이 지워지면(다 봤는데 못 본 화면) 그 화면 줄도 지워진다. 케이스 줄은 칸이 비어 이 열쇠를 안 본다
  FOREIGN KEY (service_id, env, state, screen_url) REFERENCES screen_record (service_id, env, state, url) ON DELETE CASCADE
);
CREATE UNIQUE INDEX service_api_use_case ON service_api_use (api_id, tc_id) WHERE tc_id IS NOT NULL;
CREATE UNIQUE INDEX service_api_use_screen ON service_api_use (api_id, env, state, screen_url) WHERE tc_id IS NULL;
CREATE INDEX service_api_use_tc ON service_api_use (tc_id) WHERE tc_id IS NOT NULL;
```

- **줄 하나는 케이스 줄이거나 화면 줄이다**(CHECK) — 케이스 줄은 `tc_id` · `run_id`, 화면 줄은 화면 기록 열쇠(`env` · `state` · `screen_url`)만 찬다
- 화면 줄은 화면 기록을 가리킨다 — 화면 기록이 지워지면 같이 지워진다(`ON DELETE CASCADE`). 케이스 줄은 실행을 가리키지 않는다 — 실행을 지워도 지도는 남는다
- 서비스를 지우면 같이 지워진다. API 줄 자체를 지우는 주기는 L6 와 같이 정한다([1-제품과-구조.md](1-제품과-구조.md) 「지금 만들지 않는다」)
- `grafana_ro` 에는 안 준다 — 어느 패널도 안 읽는다

사람이 적은 서비스 소개 · 정책 · 프로젝트 규칙과 첨부 문서다. **무엇을 왜 담는지는 [도메인/작성.md](../도메인/작성.md) §3.6 「★ 서비스 가이드」가 정본이다.** 여기는 표와 칸만 적는다.

```sql
CREATE TABLE service_guide_file (
  id               BIGSERIAL   PRIMARY KEY,
  service_id       BIGINT      NOT NULL REFERENCES service(id),
  name             TEXT        NOT NULL,
  size             BIGINT      NOT NULL CHECK (size >= 0),
  uploaded_by      TEXT        NOT NULL,

-- 가이드. 서비스마다 한 벌이고 바꿀 때마다(글 저장 · 파일 더하기 · 빼기) 한 행이다. 고치지 않는다 (prd_version 본보기)
CREATE TABLE service_guide_version (
  service_id    BIGINT      NOT NULL REFERENCES service(id),
  version       INTEGER     NOT NULL CHECK (version >= 1),
```

### 프로젝트 표 (`project`, 2026-10-19 · PRJ-F1-01)

애플리케이션(`service`)을 묶는 폴더다. **층이 무엇이고 왜 이렇게 나눴는지는 [공통/1-제품과-구조.md](1-제품과-구조.md) §2 「프로젝트 · 애플리케이션 · API 서버는 다른 층이다」가 정본이다.** 여기는 표와 칸만 적는다.
마이그레이션은 `db/migrations/20261019000004_project.sql` 이다.

```sql
CREATE TABLE project (
  id         BIGSERIAL   PRIMARY KEY,
  name       TEXT        NOT NULL,          -- 같은 이름을 막지 않는다. 처음 옮길 때 서비스 이름을 그대로 받기 때문이다
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 애플리케이션 하나는 프로젝트 하나에 속한다. 결과 모양이다 — 마이그레이션은 빈 칸으로 더하고 → 지금 서비스를 채우고 → NOT NULL 을 건다
ALTER TABLE service ADD COLUMN project_id BIGINT NOT NULL REFERENCES project(id);
```

- **처음 옮길 때 지금 있던 서비스는 저마다 같은 이름의 프로젝트에 든다** — 데모마켓 · 코디세이 · 데모는 서로 상관없는 사이트라 한 프로젝트에 넣으면 E2E 후보에 남의 사이트 케이스가 섞인다. 묶을 것은 사람이 옮긴다(PRJ-F1-02 · 03)
- **프로젝트를 안 주고 서비스를 만들면 DB 가 같은 이름의 프로젝트를 만들어 붙인다**(트리거 `service_default_project`). DB 검사 fixture 와 프로젝트가 아직 없을 때 첫 서비스를 컨테이너에서 만드는 `scripts/add-service.ts` 가 프로젝트를 몰라도 되게 하려는 것이다 — 서비스를 만드는 검사가 많다.
  **설정 API 는 이 트리거에 기대지 않는다** — `projectId` 를 꼭 받는다(도메인/인증 §7 `POST /api/settings/services`).
  **같은 접두사의 서비스가 이미 있으면 그 프로젝트를 쓴다** — `ON CONFLICT (prefix)` 로 끝나는 INSERT 도 트리거를 먼저 지나서, 안 그러면 같은 접두사를 다시 넣을 때마다 빈 프로젝트가 쌓인다
- 프로젝트를 비울 수 없다(`NOT NULL`). 프로젝트를 지우는 통로는 없다 — 서비스처럼 지난 실행이 가리키는 것을 흔들지 않는다. 다만 애플리케이션을 옮겨서 비게 된 프로젝트는 설정 API 가 같은 트랜잭션에서 지운다(도메인/인증 §7 `PATCH /api/settings/services/:id`의 `projectId`)
- **서비스 행을 지우면 남은 애플리케이션이 없는 그 프로젝트도 지운다**(트리거 `service_drop_empty_project`. 옮겨서 비는 경우는 트리거가 아니라 설정 API 코드가 지운다). 서비스 행을 지우는 것은 DB 검사 fixture 뿐이다(제품은 `is_active` 로 끈다) — 안 지우면 위 트리거가 만든 빈 프로젝트가 검사마다 쌓인다
- 한계 — 같은 새 접두사를 동시에 둘이 넣으면 진 쪽의 프로젝트가 빈 채로 남을 수 있다(접두사 확인과 프로젝트 만들기 사이를 잠그지 않는다). 설정에서 같은 접두사를 같은 순간에 넣어야 생긴다
- 트리거가 만든 프로젝트 이름은 그때의 서비스 이름이다 — 서비스 이름을 바꿔도 안 따라간다. 이름 바꾸기는 PRJ-F1-02 통로로 한다
- `grafana_ro` 에는 안 준다 — 어느 패널도 안 읽는다
- **권한은 프로젝트에 붙는다** — `user_project`(위 §6). 프로젝트가 지워지면(애플리케이션을 옮겨 비게 됐을 때) 그 권한 줄도 같이 지워진다(`ON DELETE CASCADE`)

### API 서버 이름 표 (`project_api_server`, 2026-10-20 · PRJ-F1-04)

호스트에 사람이 붙인 이름이다 — 케이스 목록의 API 서버 꼬리표(칩)가 이 이름을 단다. **층이 아니라 꼬리표인 까닭은 [공통/1-제품과-구조.md](1-제품과-구조.md) §2 「프로젝트 · 애플리케이션 · API 서버는 다른 층이다」가 정본이다.** 여기는 표와 칸만 적는다.
마이그레이션은 `db/migrations/20261020000001_project_api_server.sql` 이다.

```sql
CREATE TABLE project_api_server (
  project_id BIGINT NOT NULL REFERENCES project(id) ON DELETE CASCADE,
  host       TEXT   NOT NULL CHECK (host = lower(host) AND length(host) BETWEEN 1 AND 255),   -- service_api.host 와 같은 꼴(포트까지, 소문자)
  name       TEXT   NOT NULL CHECK (name = btrim(name) AND char_length(name) BETWEEN 1 AND 20),
  PRIMARY KEY (project_id, host)
);
```

- **이름은 프로젝트마다 붙는다** — 같은 인증 서버를 프로젝트 안의 애플리케이션(프론트 · 백오피스)이 같이 쓰므로 한 번 적으면 모두에 붙는다. 호스트는 `service_api.host`(작성 §3.6 「서비스 지도」)의 값 그대로다
- **같은 이름을 호스트 여럿에 써도 된다**(키는 `project_id` · `host` 이고 이름은 유일하지 않다) — 운영 · 검증 서버(`api.…` · `api-stg.…`)를 한 이름으로 묶는다
- **이름 없는 호스트는 줄이 없다** — 이름을 비우면 줄을 지운다. 표에는 이름이 있는 호스트만 든다
- 호스트가 어떤 케이스에 닿는지는 이 표가 아니라 서비스 지도(`service_api_use` 의 케이스 줄)다. 이 표는 이름만 안다
- 프로젝트가 지워지면(애플리케이션을 옮겨 비게 됐을 때) 이름 줄도 같이 지워진다(`ON DELETE CASCADE`)
- `grafana_ro` 에는 안 준다 — 어느 패널도 안 읽는다

```sql
ALTER TABLE service ADD COLUMN api_doc_env  TEXT;

CREATE TABLE api_doc_version (
  id          BIGSERIAL   PRIMARY KEY,
```

### 애플리케이션 종류 APP (2026-10-21 · APP-F1-14 · 반영 APP-F1-15 `20261021000003_app_kind.sql`)

앱 애플리케이션을 종류 「Android 앱」으로 가른다(종류 「게임」 `GAME` 은 아래 「게임 테스트 표」가 더했다 — 2026-10-22 GAME-F1-07). **무엇을 왜 하는지는 [도메인/러너.md](../도메인/러너.md) §5.2 「애플리케이션 종류 「Android 앱」」이 정본이다** (계약 블록도 거기). 여기는 칸과 한 번 옮기는 규칙만 적는다.

```sql
ALTER TABLE service DROP CONSTRAINT service_kind_check;
ALTER TABLE service ADD CONSTRAINT service_kind_check CHECK (kind IN ('SCREEN', 'APP', 'API'));

-- 한 번 옮긴다 (결정 5) — android 가 아닌 케이스가 없고, android 케이스가 있거나 대상 서버 줄에 앱 파일 경로가 있는 SCREEN 행.
-- test_case 는 서비스 번호가 없어 tc_id 접두사로 짚는다(설정 목록 caseCount 와 같은 짝짓기 — apps/admin/src/settings/store.ts).
-- caseCount 와 달리 꺼진 케이스도 센다 — 꺼 둔 웹 케이스가 있으면 웹이다. 그래서 설정의 케이스 수가 0 이어도 웹으로 남을 수 있다
UPDATE service s SET kind = 'APP'
 WHERE s.kind = 'SCREEN'
   AND NOT EXISTS (SELECT 1 FROM test_case c WHERE c.tc_id LIKE s.prefix || '-%' AND c.platforms <> '["android"]'::jsonb)
   AND (EXISTS (SELECT 1 FROM test_case c WHERE c.tc_id LIKE s.prefix || '-%')
        OR EXISTS (SELECT 1 FROM service_env e WHERE e.service_id = s.id AND coalesce(e.app_path, '') <> ''));
```

- 옮긴 뒤로는 바꾸지 않는다 — 설정 API 의 PATCH 는 그대로 400 `KIND_IMMUTABLE` 이다(도메인/인증 §7)
- **되돌리는 쪽은 `APP` 행을 먼저 `SCREEN` 으로 바꾼 뒤** 옛 CHECK `('SCREEN', 'API')` 를 건다 — 안 바꾸면 남은 `APP` 행 때문에 제약을 못 건다
- `service_api_doc_kind` 는 그대로다 — `APP` 도 API 문서 칸이 비어야 한다

### 성능 테스트 표 (2026-10-21 · PERF-F1-01)

서버 부하 · 페이지 성능 · 기기 성능의 시나리오 · 실행 · 초 단위 숫자 · 에이전트를 담는다. **무엇을 왜 하는지는 [도메인/성능.md](../도메인/성능.md) §3.9 가 정본이다.** 여기는 표와 칸만 적는다.

```sql
-- 성능 에이전트 — 부하 발생(LOAD) · 모니터링(MONITOR). 토큰은 해시만 (도메인/성능 §5.4)
CREATE TABLE perf_agent (
  id           BIGSERIAL   PRIMARY KEY,
  name         TEXT        NOT NULL UNIQUE CHECK (length(name) BETWEEN 1 AND 40),
  mode         TEXT        NOT NULL CHECK (mode IN ('LOAD', 'MONITOR')),
  project_id   BIGINT      REFERENCES project(id) ON DELETE CASCADE,   -- 모니터링 에이전트가 매인 프로젝트. 부하 발생 에이전트는 공용이라 NULL
  token_hash   TEXT        NOT NULL UNIQUE,          -- pga_ 토큰의 SHA-256 16진
  host         TEXT,                                 -- 에이전트가 알린 호스트 이름 (보이기만 한다)
  version      TEXT,                                 -- 에이전트 판
  max_vus      INTEGER     CHECK (max_vus IS NULL OR max_vus > 0),   -- LOAD 만. 비면 1,000 으로 센다 — 실측 전 임시 값이다. 기본 에이전트는 hello 의 maxVus 로 500 을 보낸다(공통/6 §9 · 도메인/성능 §5.4 실측). hello 의 maxVus 는 이 칸이 빌 때만 채우고, 사람이 PATCH 로 고친 값이 이긴다(PERF-F1-03 시작 질문)
  busy_run_id  BIGINT,                               -- 지금 맡은 실행 (외래키 없음 — 실행을 지워도 남는다)
  last_seen_at TIMESTAMPTZ,
  created_by   TEXT        NOT NULL,
  created_at   TIMESTAMPTZ NOT NULL DEFAULT now(),
  CHECK (mode = 'LOAD' OR max_vus IS NULL),
  CHECK ((mode = 'MONITOR') = (project_id IS NOT NULL))
);

-- 성능 시나리오. 정기 실행 네 칸은 service 의 네 칸과 같은 규칙이다 (공통/6 §9.2)
CREATE TABLE perf_scenario (
  id            BIGSERIAL   PRIMARY KEY,
  service_id    BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  name          TEXT        NOT NULL CHECK (length(name) BETWEEN 1 AND 100),
  version       INTEGER     NOT NULL DEFAULT 1,      -- 마지막 버전
  accounts_csv  TEXT,                                -- 테스트 계정 CSV 원문 — 비밀값. 응답에는 행 수만
  deleted_at    TIMESTAMPTZ,                         -- 지워도 실행 기록이 가리킨다
  schedule_at   TIME,
  schedule_tz   TEXT,
  schedule_env  TEXT,
  schedule_last TIMESTAMPTZ,
  created_by    TEXT        NOT NULL,
  created_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  CHECK ((schedule_at IS NULL) = (schedule_tz IS NULL) AND (schedule_at IS NULL) = (schedule_env IS NULL))
);

-- 버전마다 트랜잭션 목록 · 부하 설정 · 성능 목표 · 실행 설정을 통째로 남긴다
CREATE TABLE perf_scenario_version (
  scenario_id BIGINT      NOT NULL REFERENCES perf_scenario(id) ON DELETE CASCADE,
  version     INTEGER     NOT NULL CHECK (version >= 1),
  body        JSONB       NOT NULL,                  -- { transactions | dummyClient, load, targets, run } (도메인/성능 §3.9 「스크립트는 생성기가 만든다」)
  source      TEXT        NOT NULL CHECK (source IN ('RECORD', 'SWAGGER', 'FILE', 'MANUAL')),
  recording   TEXT,                                  -- 가린 녹화 원본 파일 이름 (artifacts/perf-record/) — RECORD 만
  created_by  TEXT        NOT NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (scenario_id, version)
);

-- 성능 실행 하나 = 화면 PRF-<id>. 서버 부하(LOAD) · 페이지 단독 측정(PAGE) · 기기 성능(DEVICE)
CREATE TABLE perf_run (
  id                BIGSERIAL   PRIMARY KEY,
  kind              TEXT        NOT NULL CHECK (kind IN ('LOAD', 'PAGE', 'DEVICE')),
  service_id        BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  scenario_id       BIGINT      REFERENCES perf_scenario(id),     -- LOAD 만. 지운 시나리오도 deleted_at 으로 남는다
  scenario_version  INTEGER,
  env               TEXT        NOT NULL,
  base_url          TEXT        NOT NULL,             -- 그때 친 주소 박제
  test_type         TEXT        CHECK (test_type IN ('LOAD', 'STRESS', 'SPIKE', 'SOAK')),
  config            JSONB       NOT NULL,             -- 부하 설정 · 성능 목표 · 에이전트 · 기기 · 측정 항목 박제 (비밀값 뺌)
  status            TEXT        NOT NULL CHECK (status IN ('QUEUED', 'SMOKE', 'RUNNING', 'DONE', 'STOPPED', 'ERROR')),
  verdict           TEXT        CHECK (verdict IN ('PASS', 'FAIL')),
  stop_reason       TEXT        CHECK (stop_reason IN ('TARGET_EXCEEDED', 'ERROR_LIMIT', 'AGENT_LOST', 'SMOKE_FAILED', 'USER')),
  targets           JSONB,                            -- [{ metric, scope, op, value, source, reqId?, measured, verdict }]
  peak_vus          INTEGER,                          -- 스트레스 테스트는 최대 동시 사용자, 그 밖은 최대 가상 사용자
  max_tps           NUMERIC(10,1),
  p95_ms            INTEGER,
  error_rate        NUMERIC(6,3),                     -- %
  degraded          BOOLEAN,                          -- 성능 저하. 견줄 이전 실행이 없으면 NULL
  ai                JSONB,                            -- AI 분석 { title, items, markers? }. 못 만들면 NULL
  origin            TEXT        NOT NULL CHECK (origin IN ('MANUAL', 'SCHEDULED')),
  triggered_by      TEXT,
  triggered_by_name TEXT,
  created_at        TIMESTAMPTZ NOT NULL DEFAULT now(),
  started_at        TIMESTAMPTZ,
  finished_at       TIMESTAMPTZ,
  CHECK ((kind = 'LOAD') = (scenario_id IS NOT NULL)),
  CHECK ((kind = 'LOAD') = (test_type IS NOT NULL)),
  CHECK ((scenario_id IS NULL) = (scenario_version IS NULL)),
  -- 끝난 모양마다 상태 · 판정 · 까닭의 짝 (도메인/성능 §3.9 「끝난 모양마다 상태 · 판정 · 까닭」 표)
  CHECK (CASE status
           WHEN 'DONE'    THEN verdict IS NOT NULL AND (stop_reason IS NULL OR (stop_reason = 'TARGET_EXCEEDED' AND verdict = 'FAIL'))
           WHEN 'STOPPED' THEN (stop_reason = 'ERROR_LIMIT' AND verdict = 'FAIL')
                            OR (stop_reason IN ('SMOKE_FAILED', 'AGENT_LOST', 'USER') AND verdict IS NULL)
           ELSE verdict IS NULL AND stop_reason IS NULL
         END)
);

-- 1초 숫자 (서버 부하 · 게임 서버 부하). 트랜잭션마다 한 행과 전체 한 행(tx = '')
CREATE TABLE perf_series (
  run_id   BIGINT    NOT NULL REFERENCES perf_run(id) ON DELETE CASCADE,
  sec      INTEGER   NOT NULL CHECK (sec >= 0),
  tx       TEXT      NOT NULL,
  vus      INTEGER   NOT NULL,
  requests INTEGER   NOT NULL,
  errors   INTEGER   NOT NULL,
  hist     INTEGER[] NOT NULL,                        -- 응답시간 분포 116칸 (도메인/성능 §3.9 「분포」)
  PRIMARY KEY (run_id, sec, tx)
);

-- 서버 모니터링 1초 숫자
CREATE TABLE perf_host_metric (
  run_id       BIGINT       NOT NULL REFERENCES perf_run(id) ON DELETE CASCADE,
  agent_id     BIGINT       NOT NULL,                 -- 모니터링 에이전트 (외래키 없음 — 지워도 남는다)
  agent_name   TEXT         NOT NULL,                 -- 그때 이름 박제
  sec          INTEGER      NOT NULL,
  cpu          NUMERIC(5,1),
  mem          NUMERIC(5,1),
  disk_io_kbps INTEGER,
  net_in_kbps  INTEGER,
  net_out_kbps INTEGER,
  PRIMARY KEY (run_id, agent_id, sec)
);

-- 페이지 측정. 단독 측정은 sec · vus 가 NULL, 부하 중 측정은 그때의 초와 가상 사용자 수
CREATE TABLE perf_page_metric (
  id          BIGSERIAL    PRIMARY KEY,
  run_id      BIGINT       NOT NULL REFERENCES perf_run(id) ON DELETE CASCADE,
  sec         INTEGER,
  vus         INTEGER,                                -- 지금은 비워 둔다 — 읽을 때 그 초의 전체 줄(perf_series tx '')에서 붙인다 (PERF-F1-07)
  url         TEXT         NOT NULL,
  device      TEXT         NOT NULL CHECK (device IN ('desktop', 'mobile')),
  network     TEXT         NOT NULL CHECK (network IN ('NONE', 'SLOW_4G')),
  lcp_ms      INTEGER,
  fcp_ms      INTEGER,
  cls         NUMERIC(5,3),
  ttfb_ms     INTEGER,
  lcp_element TEXT,
  resources   JSONB,                                  -- 로딩 워터폴 [{ name, type, startMs, durationMs, bytes }]
  ai          JSONB                                   -- 단독 측정의 페이지별 AI 분석
);

-- 기기 성능 1초 숫자. 같은 모델 두 대를 나란히 재도 겹치지 않게 일련번호로 가른다
CREATE TABLE perf_device_metric (
  run_id      BIGINT       NOT NULL REFERENCES perf_run(id) ON DELETE CASCADE,
  serial      TEXT         NOT NULL,                  -- 기기 일련번호 (adb serial · 팜 udid)
  device      TEXT         NOT NULL,                  -- 기기 이름 박제 (Galaxy A23) — 화면에 보이는 이름
  sec         INTEGER      NOT NULL,
  loop        INTEGER,                                -- 회차 (1부터)
  segment     TEXT,                                   -- 구간 이름 (로비 · 전투)
  fps         NUMERIC(5,1),
  jank_pct    NUMERIC(5,2),                           -- 프레임 드랍률
  freeze_count INTEGER CHECK (freeze_count >= 0),     -- 그 간격의 프리징(0.7초 넘게 걸린 프레임) 수. 못 읽으면 NULL (2026-10-31 WEB-F7-26)
  cpu         NUMERIC(5,1),
  mem_mb      INTEGER,
  temp_c      NUMERIC(4,1),
  battery_pct NUMERIC(5,1),
  PRIMARY KEY (run_id, serial, sec)
);

-- 실행 로그 (스모크 테스트 통과 · 400명 도달 · 에러 증가 시작 · 자동 중단)
CREATE TABLE perf_event (
  id      BIGSERIAL   PRIMARY KEY,
  run_id  BIGINT      NOT NULL REFERENCES perf_run(id) ON DELETE CASCADE,
  at      TIMESTAMPTZ NOT NULL DEFAULT now(),
  level   TEXT        NOT NULL CHECK (level IN ('INFO', 'WARN')),
  message TEXT        NOT NULL                        -- 화면에 그대로 뜨는 한 줄
);

-- AI 일. 작성 에이전트가 집는다 (도메인/성능 §7 「AI 일」)
CREATE TABLE perf_ai_job (
  id          BIGSERIAL   PRIMARY KEY,
  kind        TEXT        NOT NULL CHECK (kind IN ('DRAFT_RECORD', 'DRAFT_SWAGGER', 'DRAFT_FILE', 'SUGGEST_LOAD', 'SUGGEST_TARGETS', 'SUGGEST_PAGES', 'SUGGEST_DEVICES', 'ANALYZE_RUN', 'SUGGEST_FLOWS')),  -- SUGGEST_FLOWS 는 20261023000001_flow_suggest.sql
  service_id  BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  run_id      BIGINT      REFERENCES perf_run(id) ON DELETE CASCADE,
  input       JSONB       NOT NULL,
  status      TEXT        NOT NULL CHECK (status IN ('QUEUED', 'RUNNING', 'DONE', 'FAILED')),
  output      JSONB,
  error       TEXT,
  created_by  TEXT,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now(),
  claimed_at  TIMESTAMPTZ,
  finished_at TIMESTAMPTZ
);

-- 더미 클라이언트 — admin 이 올린 zip. 파일은 admin 만 붙이는 perf_dummy 볼륨의 <id>.zip (2026-10-23 · 도메인/성능 §7 「올리기」)
CREATE TABLE perf_dummy_client (

  service_id BIGINT      NOT NULL REFERENCES service(id) ON DELETE CASCADE,
  name       TEXT        NOT NULL,
  command    TEXT        NOT NULL,                    -- zip 안에서 띄울 명령 (./bot --users $QAHUB_USERS)
  bytes      BIGINT      NOT NULL,
  created_by TEXT        NOT NULL,
  created_at TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 실행 로그 · 결과 화면이 실행 하나의 줄을 차례로 읽는다
CREATE INDEX perf_event_run ON perf_event (run_id, id);
CREATE INDEX perf_page_metric_run ON perf_page_metric (run_id);
```

### 게임 테스트 표 (2026-10-22 · GAME-F1-02 — 반영 GAME-F1-07 · 12 · 13)

```sql
-- 반영 — db/migrations/20261022000001_game.sql (GAME-F1-07)
-- 애플리케이션 종류 GAME (게임.md §3.10 결정 6). APP 처럼 만들 때 정하고 바꾸지 않는다. 한 번 옮기는 행은 없다 — 게임 애플리케이션이 없었다
```

```sql
-- BTS 로 보낼 것. 60초 점검이 next_at 이 지난 PENDING 을 보낸다(정기 실행 틀). 1분 간격 5번 · 4xx 는 바로 GAVE_UP(408 · 429 는 다시)

  id          BIGSERIAL   PRIMARY KEY,
  project_id  BIGINT      NOT NULL REFERENCES project(id) ON DELETE CASCADE,
  kind        TEXT        NOT NULL CHECK (kind IN ('CREATE', 'COMMENT', 'TRANSITION', 'ASSIGN')),
  issue_key   TEXT,                   -- CREATE 는 보낸 뒤 채운다
  service_id  BIGINT      REFERENCES service(id) ON DELETE CASCADE,   -- 실패 등록이면 그 애플리케이션
  run_id      BIGINT      REFERENCES test_run(run_id) ON DELETE SET NULL,   -- 실패 등록이면 그 실행 — 실패 카드가 읽는다
  tc_id       TEXT,
```

```sql
-- BTS-F4-02 · F4-03 — 노트 · 일마다 MCP 열쇠 · 노트 고치기 줄
-- 프로젝트 노트. 저장할 때마다 한 행이다(prd_version 본보기). AI 가 실패하면 행이 안 생겨 앞 버전이 남는다
CREATE TABLE project_note (
  project_id  BIGINT      NOT NULL REFERENCES project(id) ON DELETE CASCADE,
  version     INTEGER     NOT NULL CHECK (version >= 1),
  sections    JSONB       NOT NULL,   -- { overview, features, recent, fragile, tips, more } — 칸마다 길이 상한은 코드 상수가 정본
  request_id  BIGINT      REFERENCES authoring_request(id) ON DELETE SET NULL,   -- 고친 NOTE 요청
  saved_at    TIMESTAMPTZ NOT NULL DEFAULT now(),
  PRIMARY KEY (project_id, version)
);

-- 노트를 고칠 사건. 열린 NOTE 요청이 없을 때 하나를 세운다 — 집을 때 그때까지 쌓인 줄에 request_id 를 적어 한 번에 넘긴다
CREATE TABLE project_note_event (
  id          BIGSERIAL   PRIMARY KEY,
  project_id  BIGINT      NOT NULL REFERENCES project(id) ON DELETE CASCADE,
  kind        TEXT        NOT NULL CHECK (kind IN ('DEPLOY', 'TASK_DONE', 'ISSUE_OPEN', 'ISSUE_CLOSE', 'PR_MERGED', 'PRD_VERSION', 'AUTHORING_DONE')),
  payload     JSONB       NOT NULL,   -- PR_MERGED 는 { issueKey, prNumber, prUrl, prTitle } — 바뀐 파일 경로는 노트 고치기 일이 PR 번호로 읽는다. prTitle 은 빈 글자일 수 있다(BTS 자동 설정 규칙 몸은 안 싣는다 — 도메인/BTS §7)
  request_id  BIGINT      REFERENCES authoring_request(id) ON DELETE SET NULL,
  created_at  TIMESTAMPTZ NOT NULL DEFAULT now()
);

-- 작성 요청 종류 NOTE 와 그 프로젝트 칸 · 일마다 MCP 열쇠
-- NOTE 는 애플리케이션이 아니라 프로젝트의 일이라 service_id 를 비우고 project_id 를 채운다(BTS.md 결정 19). 다른 종류는 지금처럼 service_id 가 찬다
ALTER TABLE authoring_request ALTER COLUMN service_id DROP NOT NULL;
ALTER TABLE authoring_request ADD COLUMN project_id BIGINT REFERENCES project(id) ON DELETE CASCADE;   -- 빈 프로젝트 지우기(service_drop_empty_project)가 노트 요청과 같이 지운다
-- 줄마다 애플리케이션 · 프로젝트 둘 중 하나만 찬다. 프로젝트 칸은 NOTE 와 그 RERUN 만 — NOT NULL 을 뗀 자리를 이 셋이 다시 막는다
ALTER TABLE authoring_request ADD CONSTRAINT authoring_request_scope_check
  CHECK ((service_id IS NULL) <> (project_id IS NULL));
ALTER TABLE authoring_request ADD CONSTRAINT authoring_request_note_scope_check
  CHECK (kind <> 'NOTE' OR project_id IS NOT NULL);
ALTER TABLE authoring_request ADD CONSTRAINT authoring_request_project_kind_check
  CHECK (project_id IS NULL OR kind IN ('NOTE', 'RERUN'));
-- kind CHECK 에 'NOTE' 를 더한다. 뿌리라 source_id 가 비고(source_pair CHECK 의 뿌리 목록에 더한다), RERUN 이 가리킬 수 있고(원본의 프로젝트 칸을 물려받는다) MERGE 는 없다
-- 자동 AI 의 MCP 열쇠 mcj_ 해시. 집을 때 새로 나고 이 줄이 RUNNING 이 아니면 무효다(BTS.md 결정 20). 받는 줄은 도메인/BTS 「열쇠 표」가 정본이다
ALTER TABLE authoring_request ADD COLUMN mcp_key_hash TEXT UNIQUE;
```

- **service_id 를 비우는 까닭** — 노트는 프로젝트마다 하나라 붙일 애플리케이션이 없다. 첫 애플리케이션에 붙이면 그 애플리케이션 권한 · 지움에 노트가 끌려간다. 작성 목록 · 결정 목록 · 대시보드처럼 `service_id` 로 거르는 질의는 NOTE 줄을 보지 않는다 — 노트 고치기는 노트 보기 화면의 「마지막 고친 때」로만 보인다
- **NOTE 줄을 집는 길** — 집기(`apps/admin/src/authoring/agentStore.ts`)는 애플리케이션마다 부른다. 그 애플리케이션 줄과 같은 프로젝트의 NOTE 줄(그 RERUN 포함)을 번호 차례로 함께 본다 — `service_id = $1 OR (service_id IS NULL AND project_id = <그 애플리케이션의 프로젝트>)`. 프로젝트 안 어느 애플리케이션 집기에서나 집히고 한 번만 집힌다(지금처럼 행 잠금)
