## 5. 계약 (변경 시 반드시 이 문서부터 수정)

### 5.1 Shared Kernel 타입

`packages/kit/src/types.ts`. 모든 컨텍스트가 이 타입만 주고받는다.

```ts
export type TcId = string;                         // 'AUTH-002'
export type ProcId = string;                       // 'MTA-TP-012'. 프로시저 번호 — 케이스 번호가 아니다 (아래 「테스트 프로시저 타입」 · 2026-11-24 TP-F2-06)
export type BrowserPlatform = 'desktop' | 'mobile';   // 브라우저 디바이스. E2E 웹 단계가 도는 창 (2026-10-28 — 앞 판은 「E2E 시나리오는 이것만 받는다」)
export type Platform = BrowserPlatform | 'android' | PcPlatform;   // android 는 Appium 으로 폰에서 도는 앱 케이스 (아래 계약 블록). PcPlatform 은 게임 윈도우 PC 다섯 값 — 아래 「윈도우 PC 게임」 (2026-11-19 GAME-F4-02)
export type ItemStatus = 'PASS' | 'FAIL' | 'NA';   // 3종을 늘리지 않는다. 늘리면 화면·문서·조회가 동시에 멈춘다.
                                                   // '아직 안 끝남'은 finished_at이 NULL인 것으로 구분한다 (§3.2)
export type JsonSchema = Record<string, unknown>;  // zod 내장 z.toJSONSchema(schema, { io: 'input' }) 출력. 검증하지 않고 그대로 저장·전달한다
export const TECHNIQUES = ['경계값 분석', '동등 분할', '결정 테이블', '상태 전이', '오류 추정'] as const;  // 설계 기법 이름. 바꾸지 않는 식별자다 (아래 계약 블록)
export type Technique = (typeof TECHNIQUES)[number];

export interface CaseSpec {
  tcId: TcId;
  name: string;
  platforms: Platform[];        // 비면 ['desktop']. android · 윈도우 값은 desktop · mobile 과 섞지 않고 android 와 윈도우 값도 같이 적지 않는다. 버튼 그림 케이스는 패드 값을 적는다 (defineCase 가 던진다 · 아래 「윈도우 PC 게임」 · 2026-11-19 반영 GAME-F4-05)
  precondition: string[];
  paramSchema: JsonSchema;      // zod 내장 변환 결과. 코드에서 null이면 빈 객체 스키마. 비밀값 칸에는 secret: true가 실려 온다 (§4.1)
  expectedSchema: JsonSchema;
  filePath: string;             // 소스 루트 기준 상대 경로
  held?: string;                // 있으면 보류 케이스, 값은 「판정 불가 — 」·「보류 — 」로 시작하는 사유 한 문장. 실행 때 건너뛴다 (아래 계약 블록)
  techniques?: Technique[];     // 그 케이스를 만든 설계 기법. 기능 테스트에만 단다. 비면 키가 없다 (아래 계약 블록)
  procId?: ProcId;              // 프로시저 안 케이스만 · 그 케이스가 든 프로시저. defineProcedure 가 싣는다 (아래 「테스트 프로시저 타입」 · 2026-11-24 TP-F2-06)
  procSeq?: number;             // 프로시저 안 케이스만 · 프로시저 안 차례(1부터)
}

export interface AssertionResult {
  statement: string;            // '응답 코드가 정상이다'
  status: ItemStatus;
  actual: unknown;
  expected: unknown;
  blocker?: boolean;            // true면 이 실패로 실행을 중단했다 (§4 verify 실패 규칙)
  notice?: string;              // 글이 글자까지 같지 않은데 통과한 문장의 알림 '기획서 문장과 글자가 다름'. PASS 에만 붙는다 (§4 「글 비교」 · 2026-11-21)
}

export interface StepResult {
  seq: number;
  tcId?: TcId;                  // 프로시저만 · 그 절차가 든 케이스 (아래 「테스트 프로시저 타입」 · 타입 2026-11-24 TP-F2-06 · kit 이 싣는다 2026-11-24 TP-F2-07)
  title: string;                // '로그인 API를 호출한다'
  status: ItemStatus;
  durationMs: number;
  startedAt?: string;           // 절차를 시작한 시각(ISO). kit 이 싣는다(건너뛴 절차는 안 싣는다). 호환성이 찍힌 화면의 단계를 찾는다 — 없는 옛 결과는 끝에서 거꾸로 센다 (도메인/호환성 §3.13 「화면 찾기」 · 2026-11-07 COMPAT-F1-08)
  assertions: AssertionResult[];
  line?: number;                // 실패한 소스 줄 번호. 코드 뷰가 이 줄을 중심으로 연다
  screenshotPath?: string;      // 실패 시 자동, 또는 capture:true인 스텝
  httpTrace?: { request: unknown; response: unknown };  // API 테스트용
  error?: { message: string; stack?: string };
  skipped?: true;               // 사람이 체크를 풀어 건너뛴 「만들기」 절차 (아래 「E2E 시나리오 타입」 · 반영 완료)
  gameFns?: string[];           // 게임 · Android 앱 항목만. 그 절차가 지난 화면 객체 함수 · 도우미 이름 (아래 「게임 테스트 타입」 · 2026-11-20 GAME-F5-06 · 앱은 APP-F4-09)
  gameFailedFns?: string[];     // 게임 · Android 앱 항목만. 그 절차를 멈춘 오류가 지나던 함수 이름 (같은 절)
}

export interface ExecuteRequest {
  runId: number;                // 스크린샷 · trace 경로 artifacts/runs/{runId}/{historyId}/ 를 만들 때와 trace 를 켤지(0 이면 끔) 정할 때만 쓴다
  historyId: number;
  tcId: TcId;
  platform: Platform;           // Playwright project 이름과 일치시킨다
  filePath: string;
  baseUrl: string;              // 이번 실행이 두드릴 주소. admin이 대상 서버 이름을 주소로 바꿔 싣는다 (§5.2)
  params: Record<string, unknown>;
  expected: Record<string, unknown>;
  timeoutMs: number;            // 기본 300000
  app?: string;                 // windows* 는 admin 이 VISION 에 깔고 받은 게임 이름(꼭 온다 — 아래 「윈도우 PC 게임」). android 는 그 대상 서버의 앱 파일 경로나 http(s) 주소(Appium 이 도는 PC 기준). 없으면 러너 환경값 PLATFORM_APP (도메인/러너 §5.2 「대상 서버마다 다른 앱 파일」 · 2026-10-19). 폰에 깔린 스토어 앱이면 패키지 이름(같은 절 「플레이스토어 앱 — 패키지 이름」 · 2026-11-04 APP-F2-01 · 반영 APP-F2-04)
  previousApp?: string;         // android 만. 실행 앱이 빌드 목록의 빌드일 때 그 줄의 바로 앞 빌드 주소 — kit 옛판에서올리기 가 깐다. 러너는 자식 PLATFORM_APP_PREVIOUS 로 넘기고 없으면 비운다 (도메인/러너 §5.2 「범용 앱 부품」 「앞 빌드」 · 2026-11-16 APP-F3-08)
  udid?: string;                // android · windows*(VISION 이 잡은 pc-<이름> · 꼭 온다) · 실제 폰 mobile(cdpUrl 이 오면 꼭 같이). admin 이 디바이스 팜에서 잡은 폰. android 는 없으면 러너 환경값 PLATFORM_DEVICE_UDID (도메인/러너 §5.2 「디바이스 팜 접속 방식」 · 2026-10-20 · TMO-F4-02)
  farmToken?: string;           // android · windows* · 실제 폰 mobile. 그 폰을 잡은 팜 토큰 — 누른 사람이나 서버 몫의 Auth 대신 PAT(2026-11-04 S25). 없으면 러너 환경값 PLATFORM_APPIUM_TOKEN(실제 폰 mobile 은 cdpUrl 이 러너 팜과 같은 호스트일 때만 — 도메인/러너 §5.2 「모바일 웹 — 실제 폰 Chrome」). 기록 · 로그에 안 넣는다 (도메인/러너 §5.2 「팜 토큰은 누른 사람 것이 먼저다」 · 2026-10-29)
  cdpUrl?: string;              // mobile 만. VISION 이 깨끗이 켜 준 그 폰 Chrome 의 DevTools 연결 주소(ws · wss). 오면 서버 흉내 대신 그 폰 Chrome 에서 돈다 (도메인/러너 §5.2 「모바일 웹 — 실제 폰 Chrome」 · 2026-11-03 TMO-F4-02)
  stability?: true;             // 정기 실행의 android 항목(Android 앱 · 게임)만. 러너가 통과해도 팜 logcat 하나를 받는다 — 안정화 지표가 통과 항목의 크래시 · ANR 도 센다. 정본은 도메인/안정화 §7 「kit」 · 「러너」 (2026-11-12 STAB-F1-04)
  uiCheck?: true;               // 호환성 실행 항목만. android 면 kit 이 capability farm:uiCheck 를 켜고 러너가 통과해도 AI 점검 결과를 받는다(COMPAT-F1-03). 모바일 웹 항목에는 안 싣는다 — admin 이 Chrome 을 켤 때 점검을 켜고 결과도 admin 이 받는다(COMPAT-F1-05). 정본은 도메인/호환성 §7 「러너 · kit」 (2026-11-07 · 타입은 COMPAT-F1-02)
  live?: string;                // 장면 열쇠(UUID — 테스트 실행은 그 시험의 trialId · 정식 실행은 항목마다 새것). 오면 마지막 장면을 GET /live/:key 로 낸다. 테스트 실행(runId 0)이면 동작마다 1초 쉬고(누른 곳 표시는 2026-11-06 WEB-F9-14 에 뺐다), 정식 실행은 장면만 찍는다 · 정식 실행 android 도 폰 장면을 찍는다(2026-11-11 — 앞 판은 400) (도메인/러너 §5.2 「테스트 모드 — 실시간 장면」 · 2026-11-03 TMO-F3-01 · 2026-11-05 WEB-F9-04)
}

  tcId?: TcId;                    // 프로시저만 · 지금 도는 케이스 (아래 「테스트 프로시저 타입」 · 타입 2026-11-24 TP-F2-06 · kit 이 싣는다 2026-11-24 TP-F2-08)
```

**테스트 프로시저가 `CaseSpec` · `StepResult` · `StepProgress` · `ExecuteRequest` · `ExecuteResponse` 를 바꾼다** (2026-11-23 · TP-F1-02).
`CaseSpec` · `StepResult` · `StepProgress` 의 새 칸과 `ProcId` 는 위 블록에 반영했다(2026-11-24 TP-F2-06 — 새 타입 `ProcedureSpec` · `ExecuteCase` · `CaseResult` 는 아래 「테스트 프로시저 타입」에만 적는다).
`ExecuteRequest` · `ExecuteResponse` 는 위 블록이 앞 꼴(케이스 하나) 그대로다 — 새 꼴은 옮기는 동안 `CasesExecuteRequest` · `CasesExecuteResponse` 라는 이름으로 같이 있고(2026-11-24 TP-F2-08 — 같은 절) admin 이 앞 꼴을 다 걷는 TP-F6-32 에 위 블록의 두 타입을 새 꼴로 바꾼다.

**`totalSteps`(분모)는 없다.** 절차는 케이스 코드가 돌면서 `seq`를 하나씩 올려 선언한다 —
끝나기 전에는 전체 개수를 Playwright도 모르고, 조건에 따라 절차 수가 갈리는 케이스도 있다.
없는 수를 지어내면 「거의 다 됐다」는 **거짓 신호**가 되고, 그것은 모르는 것보다 나쁘다.
분모를 대신하는 것은 **경과 시간과 제한 시간**이다 (§8.9).
**2026-11-06 — 분모는 kit 타입이 아니라 스캔이 준다** (사용자 · WEB-F9-13 · 도메인/실행 §3.2 「단계 목록은 처음부터 다 보인다」). 스캐너가 케이스 코드에서 글자 그대로 적힌 `test.step` 제목을 차례대로 읽어 `test_case.step_titles` 에 두고, admin 이 진행 응답에 `stepTitles` 로 싣는다.
위 「거짓 신호」 판단은 그대로 산다 — 그래서 다 셀 수 없는 케이스는 저장하지 않고(어떤 꼴인지는 도메인/카탈로그 §3.1 「단계 제목」), 도는 단계가 저장된 자리와 다르면 화면이 저장값을 버린다. 이 타입에는 `totalSteps` 를 더하지 않는다.

`RunningStep`을 화면까지 나르는 admin 조회는 여기에 `timeoutMs`를 하나 더 얹는다 (§7).
러너는 제한 시간을 모른다 — 박제된 자리가 admin의 `run_item`이다 (§7.1).

**`AssertionResult.notice` 는 「통과 · 알림」이다** (2026-11-21). `verify` 가 글끼리 견줘 글자는 다르지만 기대 글의 낱말이 화면 글에 차례대로 다 있어서 통과로 본 문장에만 붙는다.
판정 규칙은 [2-명세선언.md](2-명세선언.md) §4 「글 비교」가 정본이다. `run_item_step.assertions` 는 JSONB 라 마이그레이션이 없다.

```
[계약 변경 필요]
대상:    AssertionResult (packages/kit/src/types.ts)
현재:    statement · status · actual · expected · blocker 뿐이다. 글자가 다르지만 통과로 본 문장을 표시할 자리가 없다
```

#### E2E 시나리오 타입 (2026-09-28)

**무엇이고 왜인지는 [도메인/시나리오.md](../도메인/시나리오.md) §3.7 이 정본이다.** 여기는 admin 과 러너가 주고받는 모양만 적는다.

```ts
export type ScenarioPart =                         // 부품 목록의 정본은 도메인/시나리오 §3.7 「부품」 표
  | { kind: 'case'; tcId: TcId; params: Record<string, unknown>; expected: Record<string, unknown>;
      skipSteps: string[];                         // 건너뛸 「만들기」 절차 제목. 비면 전부 돈다
      carryOver?: boolean;                         // 넘겨받기. 없으면 true. false 면 skipSteps · links 가 비어야 한다 (결정 12)
      links?: ScenarioLink[] }                     // 이어 주기. 비면 없다 (결정 12)
  | { kind: 'api'; method: 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE'; path: string;   // '/' 로 시작. baseUrl 기준
      body?: unknown; expectStatus: number;
      service?: string;                            // 부를 애플리케이션 접두사. 없으면 시나리오 서비스. 같은 프로젝트만 (결정 13)
      auth?: string;                               // 그 단계 애플리케이션 대상 서버의 API 인증 이름. 없으면 쿠키만 간다 (결정 14)
      values?: ScenarioApiValue[];                 // 값 주입 — path · body 의 {name} 자리에 앞 단계 응답 값을 넣는다
      expectJson?: ScenarioJsonCheck[] }           // 응답 값 확인 — 응답 JSON 한 자리가 그 값과 같아야 한다
  | { kind: 'mock'; urlPattern: string; status: number; contentType: string; body: string }  // urlPattern 은 Playwright glob
  | { kind: 'unmock'; urlPattern: string }         // 같은 글자로 건 mock 을 푼다
  | { kind: 'wait'; ms: number };                  // 60000 까지

// 이어 주기 중 조립하는 사람이 거는 넷. 무엇을 · 언제만 거는지는 도메인/시나리오 §3.7 결정 12 표가 정본이다 (2026-10-06)
export type ScenarioLink =
  | { kind: 'reuse'; method: ScenarioMethod; urlPattern: string; fromSeq: number }   // 앞 응답 돌려주기
  | { kind: 'block'; method: ScenarioMethod; urlPattern: string }                    // 요청 막기. 200 {} 를 준다
  | { kind: 'rewrite'; method: ScenarioMethod; urlPattern: string;                   // 바꿔 보내기. 본문은 그대로
      to: { method: 'PUT' | 'PATCH'; path: string; value: ScenarioResponseRef } }   // path 의 {} 자리에 value
  | { kind: 'bind'; param: string; value: ScenarioResponseRef };                      // 값 꽂기. params 의 그 칸

export type ScenarioMethod = 'GET' | 'POST' | 'PUT' | 'PATCH' | 'DELETE';

export interface ScenarioApiValue {                // API 호출 단계의 값 주입 하나 (결정 14)
  name: string;                                    // path · body 의 {name} 자리. 영문 · 숫자 · 한글 · _ 1~20자
  value: ScenarioResponseRef;
}

export interface ScenarioJsonCheck {               // API 호출 단계의 응답 값 확인 하나 (결정 14)
  jsonPath: string;                                // 응답 JSON 본문의 점 표기 경로. 'items.0.title'
  equals: unknown;                                 // JSON 값. 형까지 같아야 한다
}

export interface ScenarioResponseRef {             // 앞 부품이 받은 응답에서 값 하나
  fromSeq: number;                                 // 자기보다 앞의 case 부품. API 호출 단계의 값 주입은 앞의 api 부품도 된다 (결정 14)
  method: ScenarioMethod;
  urlPattern: string;                              // 그 부품에서 이 무늬에 처음 맞은 응답
  jsonPath: string;                                // 응답 JSON 본문의 점 표기 경로. 'data.id'
}

export interface ScenarioExecuteRequest {          // POST /execute-scenario (도메인/러너 §5.2)
  runId: number | null;                            // null 이면 시험 실행. 기록이 없다
  trialId?: string;                                // 시험 실행일 때만. admin 이 만든 UUID
  platform: Platform;                              // android 는 Android 앱 · 게임이 시작 애플리케이션일 때. 웹 단계는 그때 PC 창으로 돈다 (도메인/시나리오 §3.7 결정 16)
  // 아래 다섯(app ~ adminKey)은 앱 · 게임 단계가 있을 때만 — 폰 연결이 시나리오에 하나라 요청에 한 벌이다. 뜻은 ExecuteRequest 의 같은 이름 칸과 같다 (결정 16)
  // udid · farmToken 은 실제 폰 모바일 시나리오(cdpUrl)에도 온다. 그 시나리오에는 앱 · 게임 단계가 못 낀다 — 한 폰을 Appium 과 Chrome 연결이 같이 못 쓴다 (2026-11-03 TMO-F4-02)
  app?: string;
  udid?: string;
  farmToken?: string;
  engine?: 'GAME';
  adminKey?: { header: string; value: string };   // 게임 단계 케이스에만 넘긴다. 부품 기록 · 결과에 안 넣는다
  cdpUrl?: string;                                 // mobile 시나리오만 — ExecuteRequest.cdpUrl 과 같다. 웹 단계가 그 폰 Chrome 의 창 하나 · 탭 하나로 끝까지 돈다 (결정 3)
  baseUrl: string;
  parts: Array<ScenarioPart & { filePath?: string; baseUrl?: string; apiAuths?: ApiAuth[]; apiDoc?: ApiDocRef; appStep?: boolean }>;   // case 부품에만 filePath 를 admin 이 채워 보낸다.
                                                   // baseUrl 은 case · api 부품이 돌 주소 — 시작 애플리케이션이 아닌 단계에만 붙는다. 없으면 위 baseUrl (결정 13)
```

**`StepResult` 에 `skipped?: true` 가 붙는다.** 사람이 체크를 풀어 건너뛴 「만들기」 절차다 — 판정이 없고 `PASS` 로 적되
증적이 「준비 실행 건너뜀」으로 그린다. 없는 절차로 빼 버리면 순번이 비고 「이 케이스가 준비를 세웠다」로 읽힌다.

**kit 이 새로 할 줄 알아야 하는 것** (설계서 「13. 새로 만들 것」의 kit 행).
① **케이스 본체를 꺼내 쓴다** — `defineTest` 첫 줄의 `PLATFORM_SCAN` 분기와 같은 자리에 시나리오 모드를 하나 더 낸다.
그 모드에서는 Playwright 에 등록하지 않고 본체를 넘겨준다. **`RunScope` 하나를 여러 본체에 꿴다** — 본체마다 새로 만들면 스크린샷 순번이 다시 1부터라 서로를 덮는다.
**넘겨주는 것은 본체가 아니라 그 케이스의 kit 으로 감싼 실행 함수다** (2026-09-28 구현) — 본체 안 `step()` 은 자기 kit 의 문맥을 읽으므로 다른 kit 인스턴스가 감싸면 문맥을 못 찾는다.
실행 함수는 `globalThis` 등록부에 `tcId` 로 오르고(`@platform/kit/scenario`), 앞 부품까지 쓴 순번을 받아 이어 매긴다 — 「`RunScope` 하나」는 **순번이 이어진다**는 뜻으로 지켰다.
부를 때마다 실패 표시·건너뛸 제목·결과 모으기를 새로 세우고, 본체 예외도 안에서 잡아 `{ steps, failed, error }` 로 돌려준다
② **절차를 제목으로 건너뛴다** — `step()` 이 `RunScope` 의 「건너뛸 제목」에 든 절차를 돌리지 않고 `skipped` 로 적는다.
번호는 `runStep` 이 매기므로 번호가 아니라 제목으로 지목한다. 그래서 K12 가 제목을 케이스 안에서 유일하게 가둔다([2-명세선언.md](2-명세선언.md) §4)
③ **trace 를 켰으면(`PLATFORM_TRACE=1`) 절차가 끝날 때 단계 첨부(`platform-step`)도 남긴다** (2026-10-17 E2E-F1-26) — 단독 실행과 같은 첨부다. 부품이 시작할 때도 앞 순번(첫 부품은 0)으로 하나 단다. 결과는 여전히 모은 `steps` 로 돌려주고,
첨부는 한 zip 에 섞인 부품들의 trace 를 admin 이 단계 번호로 가르는 경계로만 쓴다(도메인/시나리오 §7 「실패 기록」)
④' **앱 · 게임 단계를 돈다** (2026-10-28 E2E-F3-02, 도메인/시나리오 §3.7 결정 16) — 실행 함수가 `page` 대신 `driver`(게임이면 `game`, 어드민 키가 있으면 `adminKey` — 몸통의 `admin`)를 받고 스크린샷은 폰 화면이다.
`android` 를 선언한 케이스는 폰 여는 함수를 등록부(`platform.scenarioPhone`)에 같이 올린다 — 러너 고정 spec 이 첫 앱 단계에서 꺼내 한 번 연다. 실패한 게임 단계는 그 단계가 시작할 때의 오류 번호부터 게임 실패 파일을 남긴다

이 절의 타입은 모두 `packages/kit/src/types.ts` 에 들어 있다(2026-11-24 TP-F2-06 · 요청 · 응답은 TP-F2-08 — 더하기만 했다).
**새 꼴 요청 · 응답의 코드 이름은 옮기는 동안 `CasesExecuteRequest` · `CasesExecuteResponse` 다** (2026-11-24 사용자 · TP-F2-08 시작 질문) — admin 이 옛 꼴 케이스를 아직 앞 꼴로 보내 `ExecuteRequest` · `ExecuteResponse` 이름은 앞 꼴이 그대로 쓴다(§5.1 맨 위 타입 블록). 프로시저는 케이스 테스트 실행(TP-F2-10)과 정식 실행(2026-11-24 TP-F2-11)이 새 꼴로 보낸다.

`StepResult.tcId` 는 kit 이 싣고 `CaseResult` 는 kit 이 케이스마다 정한다(2026-11-24 TP-F2-07 — `packages/kit/src/runtime/procedure.ts`).
kit 은 그 `CaseResult` 를 케이스가 끝나는 대로 밖으로 내고 진행 줄에 `StepProgress.tcId` 를 싣는다(2026-11-24 TP-F2-08).

```ts
export type ProcId = string;                       // 'MTA-TP-012'. <접두사>-TP-<3자리>. 케이스 번호가 아니다

// CaseSpec 에 더하는 칸 — 프로시저 안 케이스만. defineProcedure 가 케이스 선언에 싣고 스캐너는 그대로 읽는다
//   CaseSpec.procId?: ProcId               그 케이스가 든 프로시저
//   CaseSpec.procSeq?: number              프로시저 안 차례(1부터)

export interface ProcedureSpec {                   // 스캐너가 프로시저 파일 하나에서 읽는 것 — export const procedure
  procId: ProcId;
  name: string;
  platforms: Platform[];        // android 하나이거나 윈도우 값. 안 케이스와 같다 (K21)
  precondition: string[];
  filePath: string;
  cases: CaseSpec[];            // 도는 차례대로. 케이스마다 procId · procSeq 가 실려 있다
}

// StepResult · StepProgress 에 더하는 칸 — 프로시저만
//   StepResult.tcId?: TcId                 그 절차가 든 케이스. 케이스 사이 절차는 다음 케이스다(그 케이스의 준비 구간)
//   StepProgress.tcId?: TcId               지금 도는 케이스. 진행 화면이 케이스를 짚는다

export interface ExecuteCase {                     // 요청의 케이스 한 건
  tcId: TcId;
  params: Record<string, unknown>;
  expected: Record<string, unknown>;
}

export interface CasesExecuteRequest {             // 새 꼴 — 파일 하나 + 케이스 목록. TP-F6-32 에 ExecuteRequest 가 된다
  runId: number;
  historyId: number;            // 케이스 줄 번호가 아니라 실행 단위 번호(run_exec.exec_id) — 아래 「번호 이름」
  platform: Platform;
  filePath: string;
  baseUrl: string;
  procId?: ProcId;              // 프로시저 파일이면 싣는다. 옛 꼴 파일은 없다
  cases: ExecuteCase[];         // 도는 차례대로. 프로시저는 그 프로시저의 케이스 전부, 옛 꼴 파일은 1건
  timeoutMs: number;            // 러너가 프로세스를 끊는 한도 — 그 실행 단위 케이스 줄 한도의 합 (도메인/러너 §5.2 「프로시저 실행」 「케이스별 한도」)
  // app · previousApp · udid · farmToken · cdpUrl · stability · uiCheck · live · engine · gamePopups · adminKey · apiAuths · apiDoc · recordHar 는 그대로다 — 실행 단위에 한 벌
}

export interface CaseResult {                      // 응답의 케이스 한 건
  tcId: TcId;
  status: ItemStatus;
  durationMs: number;
  startedAt?: string;           // 그 케이스의 시계가 시작한 때(ISO) — 준비 구간 시작. 못 간 케이스는 없다
  finishedAt?: string;          // 케이스 구간이 끝난 때. 돌다 끊긴 케이스 · 못 간 케이스는 없다
  steps: StepResult[];
  error?: { message: string; stack?: string };     // Blocked 는 message 가 'BLOCKED'
  blockedBy?: TcId;             // Blocked 일 때만. 막은 케이스
  elementChecks?: ElementCheck[];   // 게임 항목만. 그 케이스가 남긴 것
  apiCalls?: ApiCallSeen[];         // 게임 항목만. 그 케이스(준비 구간 포함)가 부른 것
}

export interface CasesExecuteResponse {            // 새 꼴. TP-F6-32 에 ExecuteResponse 가 된다
  historyId: number;            // 요청의 그 번호
  durationMs: number;
  results: CaseResult[];        // 요청 cases 와 같은 수 · 같은 차례. 못 간 케이스도 한 건씩 있다
  error?: { message: string; stack?: string };     // 케이스 판정 밖의 사유 — 프로세스가 통째로 끊겼거나(TIMEOUT · ABORTED) 판정이 다 난 뒤에 실패로 끝났다 (도메인/러너 §5.2 「프로시저 실행」). 러너 고장은 지금처럼 500 이다
  device?: DeviceInfo;
  fps?: FpsSample[];            // 게임 정식 실행만. 표본의 seq(절차 번호)로 케이스에 붙인다
}
```

- **`ItemStatus` 는 그대로 셋이다.** Blocked 는 `NA` + `error.message: 'BLOCKED'` + `blockedBy` 다(글자는 `packages/kit/src/runtime/protocol.ts` 의 `BLOCKED` 한 자리에 둔다 — kit · 러너 · admin 이 같은 글로 알아본다) — 판정값을 늘리면 화면 · 문서 · 조회가 동시에 멈춘다는 위 판단이 그대로 산다. 화면 · 리포트는 이 사유로 「Blocked」를 따로 그리고 따로 센다
- **요청은 한 꼴이다** — 파일 하나 + 케이스 목록. 옛 꼴 파일은 케이스 1건짜리 목록이다. 앞 꼴(최상위 `tcId` · `params` · `expected` · 응답 최상위 `status` · `steps`)은 옮기는 동안 러너가 같이 받고 TP-F6-32 에 걷는다. 러너는 요청에 `cases` 가 있으면 새 꼴로 답한다
- **사람이 `npx playwright` 로 직접 돌리는 납작한 입력 꼴은 남긴다** — `PLATFORM_PARAMS` 에 케이스 하나의 값을 주는 지금 길이다. 프로시저를 손으로 돌리면 케이스마다 스키마 기본값으로 돈다
- **응답에 묶은 판정을 두지 않는다** — 실행 단위의 상태는 케이스 결과에서 센다. 두 곳에 적으면 어긋난다
- **번호 이름** — 새 꼴의 `historyId` 는 실행 단위 번호다. 옛 꼴 케이스는 그 케이스 줄 번호(`run_item.history_id`)와 같은 값이라 달라지는 것이 없고, 프로시저 실행 단위만 새 번호를 받는다. 같은 번호 줄에서 뽑아 러너가 지금처럼 번호 하나로 자식 프로세스 · 중단 · 진행 · 장면 · 팜 증적 · 폴더(`artifacts/runs/{runId}/{historyId}/`)를 가른다 — 호환성의 `compat_run_case.id` 와 같은 방식이다.
  이름을 `execId` 로 바꾸지 않는다 — `/abort` · `/progress` · `/live` · 팜 증적 · 폴더 규칙이 이 이름에 묶여 있어 바꾸면 그 계약이 다 같이 바뀐다. 표는 [4-데이터모델.md](4-데이터모델.md) 「테스트 프로시저 표」
- **게임 기록은 케이스에 붙인다** — `elementChecks` · `apiCalls` 는 케이스 결과에 싣고, 러너가 1초마다 읽는 `fps` 는 표본의 절차 번호로 admin 이 케이스에 가른다
- E2E 시나리오 요청(`ScenarioExecuteRequest`)이 앱 · 게임 단계에서 프로시저를 어떻게 싣는지는 위 「E2E 시나리오 타입」 「프로시저 단계」다(TP-F1-04 — 정본은 [도메인/시나리오.md](../도메인/시나리오.md) §3.7 결정 17)

`testKind` 는 `'UI'` · `'FN'`(실행 상세의 `kind` 와 같은 칸 — 2026-10-19 ⑫ · 점검 5.6)이고 끝난 호환성 실행은 `'COMPAT'` 이다(그때 `counts` 대신 `compat: { pending, unchecked }`). 앞 판은 화면이 서비스마다 `GET /api/runs` 첫 쪽을 읽어 케이스 실행만 알렸다 — 작성과 E2E 가 안 보였다. `remainingMs` 는 도메인/실행 §8.9 「남은 시간」, `expected` 는 도메인/작성 §7 「남은 시간」과 같은 셈이다 |
| `env` · `base_url` | 요청은 **키만** 싣는다. 주소는 **그 서비스의** `service_env` 표(§6)에서 찾아 `test_run.base_url`에 박는다 | 요청이 주소를 실으면 아무 데나 쏠 수 있다. 매핑은 바뀌므로 그날 친 주소는 따로 남겨야 한다 |
| `repeat` | 요청 **최상위에 1개**. 기본 1. `repeat` 자체에는 상한을 두지 않고 **만들어질 실행 항목 건수**로 막는다 (2026-09-17 수정) | 목적은 새로 쓴 테스트 코드가 안정적인지 여러 번 돌려 보는 것이다. 항목마다 두면 `run_item` 개수 계산이 항목별로 갈라져 §8.2의 안내 문구를 못 만든다. **상한을 「두지 않는다」였던 것을 고쳤다** — 근거였던 동시 실행 2는 *돌아가는 프로세스 수*를 막지 *INSERT 건수*를 막지 않는다. 디바이스를 둘씩 선언한 케이스 10건이면 디바이스 수의 합이 20이고, `repeat` 10만이면 `run_item` 200만 행이 요청 한 번에 들어간다. **상한의 자리와 숫자도, 건수를 세는 식도 §8.2가 정본이다** (2026-09-21에 예시를 §8.2의 합 식에 맞췄다 — 앞 판의 `케이스 10 × 디바이스 2 × repeat 10만`은 모든 케이스의 디바이스 수가 같을 때만 맞는 곱셈이었다) |
| 자동 재시도 | Playwright의 `retries`는 **0으로 고정한다.** 켜려면 SPEC 변경이 먼저다 | 실패를 가리려고 다시 돌리는 것과, 실패를 드러내려고 여러 번 돌리는 것은 다른 일이다. 반복은 `repeat`가 한다 |
| `POST /api/runs/:runId/abort` | 대기 중 항목은 줄에서 빼고 `NA` + 사유로 닫고, **진행 중 항목도 끊는다.** `test_run.status = ABORTED` | `ABORTED`가 스키마에 있는데 쓰는 코드가 한 줄도 없었다. 대기만 취소하면 5분짜리가 계속 돌아 사용자에게는 멈추지 않은 것으로 보인다 |
| 중단과 결과의 경합 | 결과 되쓰기 UPDATE에 `AND finished_at IS NULL`을 붙인다 | 중단이 닫은 항목에 뒤늦게 도착한 결과가 덮어쓰면 무엇이 진짜인지 알 수 없다. **먼저 온 것만** 기록한다 |
| admin 재기동 복구 | 부팅 직후 `status = 'RUNNING'`인 실행의 미완 항목을 닫고 실행을 `ABORTED`로 바꾼다. 중단과 같은 처리다 | 디스패처는 메모리에만 있다. 재기동하면 이어갈 주체가 없는데 `RUNNING`으로 남아 영원히 진행 중인 것처럼 보인다 |
| 서비스 이름 · 테스트 저장소 | 요청은 서비스를 싣지 않는다. 서버가 `tcId` 접두사로 서비스를 알아내 `test_run.service_id`와 스냅샷 `service_name`·`tests_repo`에 박는다. 플랫폼이 저장소에서 직접 받아오지는 않는다 | 서비스마다 저장소 칸이 따로 있다. "어디서 무엇을 시험한 것인가"가 증적 문서에 남아야 한다. 접두사가 섞인 요청은 400이다 — **한 실행은 한 서비스**다 (§7 인증 적용 범위) |
| 검색 조건 | 케이스 속성(§8.1 검색 조건 표에서 거르는 곳이 「서버」인 줄)은 **서버가**, `마지막 결과`는 **화면이** 거른다(**2026-10-19 ⑫ 부터 마지막 결과도 서버가 거른다** — 실행 쪽 함수가 판정을 주고 카탈로그 질의는 `run_item` 을 안 읽는다. 정본은 도메인/카탈로그 §7 `?last=`). 서비스는 조건이 아니라 **애플리케이션 줄(앞 이름 헤더 서비스 고르개)의 선택**이고 서버가 늘 적용한다 (고르개의 정본은 공통/5-화면공통 §8) (2026-09-17 수정 — 목록이 한 서비스로 좁혀지면 그 안의 접두사는 전부 같아서 거를 것이 없다). 조건 정의는 §8.1이 정본이다 | 마지막 결과는 실행할 때마다 바뀐다. 카탈로그가 `run_item`을 읽으면 컨텍스트 경계가 깨진다 (§3.1) |
| 검색 응답 | `sort`와 `totalIsExact`를 같이 준다 | 나중에 검색엔진을 붙이면 정렬이 관련도순으로 바뀌고 건수가 근사치가 된다. 화면이 지금 모양을 전제하면 그때 깨진다 (§5.3) |
| 로그인 | 아이디·비밀번호 1종. **권한은 운영 여부 + 프로젝트마다 기능 셋 × 세 칸 + 대시보드 한 칸**, 계정은 회원가입 → admin 수락. 팀·실행 승인 절차는 만들지 않는다 (2026-09-17 수정 · 2026-09-28 기능별 권한) | 공용 비밀번호 하나는 누가 눌렀는지 못 가린다. 권한 구분은 설정 화면(§8.8)이 생기면서 필요해졌고, 설치형 납품에서 회사마다 역할이 달라 등급 셋이 기능별 칸으로 바뀌었다. 뒤집은 근거는 §3.5 와 §6의 「권한 칸」 문단에 남겨 뒀다 |

`POST /api/runs`가 실행이 끝날 때까지 기다리지 않는 이유도 같은 자리에 적어 둔다.
케이스 1건에 최대 5분을 줄 수 있으므로 기다리면 화면이 그만큼 멈춘다.
끝났는지는 `GET /api/runs/:runId`의 `status`가 `FINISHED`인지로 본다 (§3.2 불변식).

**인증 이전에 쌓인 실행의 실행자는 믿을 수 없다** (2026-09-16 결정).
로그인이 없던 동안 `triggered_by`에는 요청이 적어 보낸 값이나 기본값 `'admin'`이 들어갔다.
그 값을 사람 이름처럼 보여주면 증적 문서가 없는 근거를 만들어 낸다.

**구분 규칙.** `test_run.triggered_by_name`이 `NULL`이면 인증 도입 이전 실행이다.
화면과 증적 문서의 실행자 칸에 **`실행자 미상 (인증 도입 이전)`** 이라고 쓴다.
실행 시각으로 가르지 않는다. 마이그레이션 적용 시각과 배포 시각이 어긋나면
경계에 걸친 실행을 잘못 분류한다.
`triggered_by`에 남아 있는 옛 값은 지우지 않고 그대로 두되 사람 이름 자리에 쓰지 않는다.
지우면 그 자리에 무엇이 있었는지조차 알 수 없어진다.
